#!/usr/bin/env bash

set -Eeuo pipefail
umask 077

SUBMIT_TREE=/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2265-cohort2-perf6/submit-tree
DESIGN_REPO=/work/1/SFC/tanab/izanagi
LEDGER_ROOT=/home/SFC/tanab/.claude/jobs
EXPECT_REPO_HEAD=8bdf173cc81e5371db7b7bcddb8bdcbb5aeff235
EXPECT_CCBENCH_HEAD=511c9538e4e8efa54b45cda62e72389ed3b706ec
PBS_SCRIPT="$SUBMIT_TREE/tools/pegasus/probes/t2187_adaptive_const_probe.pbs"

fail_preflight() {
  echo "submit preflight failed: $*" >&2
  exit 2
}

fail_submit() {
  echo "submission stopped: $*" >&2
  exit 3
}

accepted_job_unrecorded() {
  printf '*** ACCEPTED JOB MAY BE UNRECORDED: rep=%s jobid=%s ***\n' "$1" "$2" >&2
  fail_submit "$3"
}

accepted_qsub_output_unrecorded() {
  printf '%s\n' '*** ACCEPTED JOB MAY BE UNRECORDED ***' >&2
  printf 'rep=%s qsub stdout follows:\n' "$1" >&2
  printf '%s\n' "$2" >&2
  printf '%s\n' '*** END QSUB STDOUT ***' >&2
  fail_submit "$3"
}

command -v qsub >/dev/null 2>&1 || fail_preflight "qsub is not available"

[[ -n "${IZ_LEDGER:-}" ]] || fail_preflight "IZ_LEDGER is required"
[[ -n "${IZ_DESIGN_COMMIT:-}" ]] || fail_preflight "IZ_DESIGN_COMMIT is required"
[[ -n "${IZ_DESIGN_SHA256:-}" ]] || fail_preflight "IZ_DESIGN_SHA256 is required"
[[ -n "${IZ_DESIGN_PATH:-}" ]] || fail_preflight "IZ_DESIGN_PATH is required"

[[ "$IZ_LEDGER" == /* && "$IZ_LEDGER" != *$'\t'* && "$IZ_LEDGER" != *$'\n'* ]] ||
  fail_preflight "IZ_LEDGER must be an absolute single-line path without tabs"
canonical_ledger=$(realpath -m -- "$IZ_LEDGER" 2>/dev/null) ||
  fail_preflight "IZ_LEDGER cannot be canonicalized"
[[ "$canonical_ledger" == "$IZ_LEDGER" && "$IZ_LEDGER" == "$LEDGER_ROOT/"* ]] ||
  fail_preflight "IZ_LEDGER must be a canonical absolute path below $LEDGER_ROOT"
[[ "$IZ_DESIGN_COMMIT" =~ ^[0-9A-Fa-f]{40}$ ]] ||
  fail_preflight "IZ_DESIGN_COMMIT must be 40 hexadecimal characters"
[[ "$IZ_DESIGN_SHA256" =~ ^[0-9A-Fa-f]{64}$ ]] ||
  fail_preflight "IZ_DESIGN_SHA256 must be 64 hexadecimal characters"
[[ "$IZ_DESIGN_PATH" != /* && "$IZ_DESIGN_PATH" != *$'\t'* && "$IZ_DESIGN_PATH" != *$'\n'* ]] ||
  fail_preflight "IZ_DESIGN_PATH must be a single-line repo-relative path without tabs"
[[ "/$IZ_DESIGN_PATH/" != */../* && "/$IZ_DESIGN_PATH/" != */./* && "$IZ_DESIGN_PATH" != */ ]] ||
  fail_preflight "IZ_DESIGN_PATH must not contain empty, dot, or dot-dot components"

git -C "$DESIGN_REPO" cat-file -e "${IZ_DESIGN_COMMIT}^{commit}" 2>/dev/null ||
  fail_preflight "IZ_DESIGN_COMMIT does not name an existing commit"
design_hash_line=$(git -C "$DESIGN_REPO" show "${IZ_DESIGN_COMMIT}:${IZ_DESIGN_PATH}" |
  sha256sum) || fail_preflight "cannot hash IZ_DESIGN_PATH at IZ_DESIGN_COMMIT"
observed_design_sha256=${design_hash_line%% *}
[[ "$observed_design_sha256" == "${IZ_DESIGN_SHA256,,}" ]] ||
  fail_preflight "IZ_DESIGN_SHA256 does not match IZ_DESIGN_PATH at IZ_DESIGN_COMMIT"

canonical_submit_tree=$(realpath -e -- "$SUBMIT_TREE" 2>/dev/null) ||
  fail_preflight "SUBMIT_TREE does not resolve"
[[ "$canonical_submit_tree" == "$SUBMIT_TREE" ]] ||
  fail_preflight "SUBMIT_TREE is not its canonical path"

repo_head=$(git -C "$SUBMIT_TREE" rev-parse HEAD) ||
  fail_preflight "cannot read SUBMIT_TREE HEAD"
[[ "$repo_head" == "$EXPECT_REPO_HEAD" ]] ||
  fail_preflight "SUBMIT_TREE HEAD mismatch"

repo_status=$(git -C "$SUBMIT_TREE" status --porcelain --untracked-files=no) ||
  fail_preflight "cannot read SUBMIT_TREE status"
[[ -z "$repo_status" ]] || fail_preflight "SUBMIT_TREE is not tracked-clean"

ccbench_head=$(git -C "$SUBMIT_TREE/external/ccbench" rev-parse HEAD) ||
  fail_preflight "cannot read ccbench HEAD"
[[ "$ccbench_head" == "$EXPECT_CCBENCH_HEAD" ]] ||
  fail_preflight "ccbench HEAD mismatch"

ccbench_status=$(git -C "$SUBMIT_TREE/external/ccbench" status --porcelain --untracked-files=all) ||
  fail_preflight "cannot read ccbench status"
[[ -z "$ccbench_status" ]] || fail_preflight "ccbench is not clean"

for required_file in \
  "$PBS_SCRIPT" \
  "$SUBMIT_TREE/tools/pegasus/probes/t2187_adaptive_const_probe.py" \
  "$SUBMIT_TREE/tools/pegasus/policy.json"; do
  [[ -f "$required_file" && ! -L "$required_file" ]] ||
    fail_preflight "required file is absent, non-regular, or a symlink: $required_file"
done

[[ ! -e "$IZ_LEDGER" && ! -L "$IZ_LEDGER" ]] ||
  fail_preflight "IZ_LEDGER already exists"

if ! (set -o noclobber; printf '%s\t%s\n' \
  submit_tree "$SUBMIT_TREE" \
  repo_head "$EXPECT_REPO_HEAD" \
  ccbench_head "$EXPECT_CCBENCH_HEAD" \
  elapstim_req 01:00:00 \
  design_commit "$IZ_DESIGN_COMMIT" \
  design_path "$IZ_DESIGN_PATH" \
  design_sha256 "${IZ_DESIGN_SHA256,,}" \
  columns $'submitted_utc\trep_index\tpbs_jobid\texpected_json\tqsub_argv' \
  > "$IZ_LEDGER") 2>/dev/null; then
  fail_preflight "cannot create IZ_LEDGER with complete metadata"
fi

REP_INDEXES=(0 1 2 3 4 5 6)
CELL_ORDERS=(
  'none:0:100:1000:10+stock:1:100:1000:10+tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1'
  'stock:1:100:1000:10+tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10'
  'tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10+stock:1:100:1000:10'
  'tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10+stock:1:100:1000:10+tuned:1:1:1000:2560'
  'cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10+stock:1:100:1000:10+tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240'
  'cw-as:1:1:1000:2560:10000:10240:1:1:4:0+cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10+stock:1:100:1000:10+tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0'
  'cw-as-dyn:1:1:1000:2560:10000:10240:1:1:4:1+none:0:100:1000:10+stock:1:100:1000:10+tuned:1:1:1000:2560+tuned-u10240:1:1:1000:10240+cw:1:1:1000:2560:10000:10240:0:100:100:0+cw-as:1:1:1000:2560:10000:10240:1:1:4:0'
)

cd "$SUBMIT_TREE"

recheck_submit_tree() {
  local current_head current_status
  current_head=$(git rev-parse HEAD) ||
    fail_submit "cannot re-read SUBMIT_TREE HEAD immediately before qsub"
  [[ "$current_head" == "$EXPECT_REPO_HEAD" ]] ||
    fail_submit "SUBMIT_TREE HEAD changed before qsub"
  current_status=$(git status --porcelain --untracked-files=no) ||
    fail_submit "cannot re-read SUBMIT_TREE status immediately before qsub"
  [[ -z "$current_status" ]] ||
    fail_submit "SUBMIT_TREE is not tracked-clean immediately before qsub"
}

for array_index in "${!REP_INDEXES[@]}"; do
  rep_index=${REP_INDEXES[$array_index]}
  cells=${CELL_ORDERS[$array_index]}
  qsub_env="IZANAGI_T2187_MODE=performance,IZANAGI_T2187_BACKOFF_TRACE=0,IZANAGI_T2187_EXTIME=6,IZANAGI_T2187_CELLS=${cells},IZANAGI_T2187_WORKLOADS=write-heavy+balanced+read-heavy,IZANAGI_T2187_THREADS=6+12+18+24+30+36+42+48,IZANAGI_T2187_REP_INDEX=${rep_index},IZANAGI_T2187_STAGE=1,IZANAGI_T2187_OUT_DIR=/work/1/SFC/tanab/izanagi-job-evidence/dynamic-backoff/perf"
  qsub_argv="qsub -l elapstim_req=01:00:00 -v '${qsub_env}' ${PBS_SCRIPT}"

  recheck_submit_tree
  if qsub_output=$(qsub -l elapstim_req=01:00:00 -v "$qsub_env" "$PBS_SCRIPT"); then
    :
  else
    qsub_rc=$?
    fail_submit "qsub failed for rep $rep_index with rc=$qsub_rc"
  fi
  qsub_output_pattern='^Request ([^[:space:]]+) submitted to queue: gen_S\.$'
  [[ "$qsub_output" =~ $qsub_output_pattern ]] ||
    accepted_qsub_output_unrecorded "$rep_index" "$qsub_output" \
      "qsub stdout did not match the measured submission format for rep $rep_index"
  jobid=${BASH_REMATCH[1]}
  [[ "$jobid" =~ ^([0-9]+:)?[A-Za-z0-9._-]+$ ]] ||
    accepted_qsub_output_unrecorded "$rep_index" "$qsub_output" \
      "qsub returned a malformed RequestID for rep $rep_index"

  submitted_utc=$(date -u +%Y-%m-%dT%H:%M:%SZ) ||
    accepted_job_unrecorded "$rep_index" "$jobid" \
      "cannot obtain submission time after qsub accepted job $jobid"
  expected_job_component=${jobid//:/_}
  # 0_ は、既存成果物 20 件すべてが 0: 接頭辞である実測に基づく。
  expected_json="/work/1/SFC/tanab/izanagi-job-evidence/dynamic-backoff/perf/stage1-rep${rep_index}-0_${expected_job_component}.json"
  if ! printf '%s\t%s\t%s\t%s\t%s\n' \
    "$submitted_utc" "$rep_index" "$jobid" "$expected_json" "$qsub_argv" \
    >> "$IZ_LEDGER"; then
    accepted_job_unrecorded "$rep_index" "$jobid" \
      "cannot record accepted job $jobid in IZ_LEDGER"
  fi
  printf 'submitted rep=%s jobid=%s\n' "$rep_index" "$jobid"
done
