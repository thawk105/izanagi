"""M-B: t080 stub-free e2e の phase 内訳を測る使い捨て plugin。

`-n 0` の単一 process で、test module の helper と subprocess/shutil を包み、
呼び出しごとの所要と入れ子を JSONL に書く。repo には入れない。
"""
import functools
import json
import os
import shutil
import subprocess
import sys
import time

_OUT = os.environ.get("IZ_PHASE_OUT")
_STACK = []
_SEQ = [0]


def _write(obj):
    if not _OUT:
        return
    with open(_OUT, "a", encoding="utf-8") as fh:
        fh.write(json.dumps(obj, sort_keys=True, default=str) + "\n")


def _summ(name, args, kwargs):
    if name == "subprocess.run":
        argv = args[0] if args else kwargs.get("args")
        if isinstance(argv, (list, tuple)):
            head = [str(a) for a in argv[:3]]
            if head and head[0] == sys.executable and len(argv) > 1:
                head = ["<python>", *[str(a) for a in argv[1:3]]]
            return " ".join(head)[:120]
        return str(argv)[:120]
    if name in ("shutil.copytree",):
        src = args[0] if args else kwargs.get("src")
        return str(src)[-90:]
    if name in ("_run_git", "_run_git_bytes"):
        return " ".join(str(a) for a in args[1:4])[:120]
    return ""


def _wrap(name, fn, current=None):
    @functools.wraps(fn)
    def wrapped(*args, **kwargs):
        _SEQ[0] += 1
        seq = _SEQ[0]
        parent = _STACK[-1][0] if _STACK else None
        depth = len(_STACK)
        t0 = time.time()
        _STACK.append((seq, name))
        child_time = [0.0]
        try:
            return fn(*args, **kwargs)
        finally:
            _STACK.pop()
            dt = time.time() - t0
            if _STACK:
                pass
            _write({
                "ev": "call", "seq": seq, "parent": parent, "depth": depth,
                "name": name, "dur": round(dt, 4), "arg": _summ(name, args, kwargs),
                "test": current[0] if current else None, "t0": t0,
            })
    return wrapped


_CURRENT = [None]


def pytest_collection_finish(session):
    mod = sys.modules.get("test_s8b_oracle_driver")
    if mod is None:
        for key, value in sys.modules.items():
            if key.endswith("test_s8b_oracle_driver"):
                mod = value
                break
    _write({"ev": "module", "found": mod is not None, "name": getattr(mod, "__name__", None)})
    if mod is None:
        return
    for attr in (
        "_build_t080_stub_free_e2e_repo", "_t080_stub_free_e2e_repo",
        "_copy_git_visible_output", "_git_visible_output_paths",
        "_run_git", "_run_git_bytes", "_copy_t080_basis_file",
        "_copy_t080_migration_basis_file", "_assert_exact_t080_live_held_markers",
    ):
        if hasattr(mod, attr):
            setattr(mod, attr, _wrap(attr, getattr(mod, attr), _CURRENT))
    migration = getattr(mod, "migration", None)
    if migration is not None:
        for attr in ("verify_receipt", "inspect_receipt_history", "draft_receipt",
                     "finalize_receipt", "validate_draft"):
            if hasattr(migration, attr):
                setattr(migration, attr, _wrap("migration." + attr, getattr(migration, attr), _CURRENT))
    shutil.copytree = _wrap("shutil.copytree", shutil.copytree, _CURRENT)
    subprocess.run = _wrap("subprocess.run", subprocess.run, _CURRENT)


def pytest_runtest_logstart(nodeid, location):
    _CURRENT[0] = nodeid
    _write({"ev": "test_start", "nodeid": nodeid, "t": time.time()})


def pytest_runtest_logreport(report):
    _write({"ev": "report", "nodeid": report.nodeid, "when": report.when,
            "duration": report.duration, "outcome": report.outcome})


def pytest_sessionfinish(session, exitstatus):
    _write({"ev": "sessionfinish", "t": time.time(), "exitstatus": int(exitstatus)})
