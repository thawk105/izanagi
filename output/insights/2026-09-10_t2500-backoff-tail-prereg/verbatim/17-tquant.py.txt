import math


def _betacf(a, b, x):
    MAXIT, EPS, FPMIN = 300, 3e-16, 1e-300
    qab, qap, qam = a + b, a + 1.0, a - 1.0
    c = 1.0
    d = 1.0 - qab * x / qap
    if abs(d) < FPMIN:
        d = FPMIN
    d = 1.0 / d
    h = d
    for m in range(1, MAXIT + 1):
        m2 = 2 * m
        aa = m * (b - m) * x / ((qam + m2) * (a + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        h *= d * c
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        de = d * c
        h *= de
        if abs(de - 1.0) < EPS:
            break
    return h


def betai(a, b, x):
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    lbeta = math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b)
    bt = math.exp(lbeta + a * math.log(x) + b * math.log(1.0 - x))
    if x < (a + 1.0) / (a + b + 2.0):
        return bt * _betacf(a, b, x) / a
    return 1.0 - bt * _betacf(b, a, 1.0 - x) / b


def t_cdf(t, nu):
    x = nu / (nu + t * t)
    p = 0.5 * betai(nu / 2.0, 0.5, x)
    return 1.0 - p if t > 0 else p


def t_ppf(p, nu):
    lo, hi = 0.0, 200.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if t_cdf(mid, nu) < p:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def u_flat(cv, h, alpha_per_limit, nu=8.0):
    v = cv * cv / 5.0
    se = math.sqrt(2.0 * v) / h
    t = t_ppf(1.0 - alpha_per_limit, nu)
    return t, 1.0 - math.exp(-t * se * math.log(2.0))


CV = 0.006151493748
H_HALF_OCT_MIN = 0.346422567
H_HALF_OCT_MAX = 0.346724613
H_SHORT = 0.11147

for label, alpha in (("0.05/18", 0.05 / 18), ("0.05/24", 0.05 / 24), ("0.05/36", 0.05 / 36)):
    t1, u1 = u_flat(CV, H_HALF_OCT_MAX, alpha)
    _, u2 = u_flat(CV, H_HALF_OCT_MIN, alpha)
    _, u3 = u_flat(CV, H_SHORT, alpha)
    print(f"alpha={label} t={t1:.6f} U_flat(half-octave)={u1*100:.4f}%..{u2*100:.4f}% "
          f"U_flat(short h={H_SHORT})={u3*100:.4f}%")
