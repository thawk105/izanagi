#!/usr/bin/env python3
"""T-1878 段 6 fix (R-2 補完): 拡張子 allowlist を外し、50 MiB 以下の全 regular file の本文と、全 entry の path 名で 934607 を探す。"""
import os
from collections import Counter

ROOTS = [
    "/work/1/SFC/tanab/dev-wave-jobs/t755-mocc-trace-execution",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-mocc-trace-execution",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-q2-mocc-trace",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-q2-mocc-trace-continuation",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1641-t755-report-binding",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1506-mocc-trace0",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1582-mocc-trace0-pilot",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-mocc-trace-pair",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-mocc-g2-repro-20260826",
    "/work/1/SFC/tanab/izanagi-job-evidence",
]
MAX_BYTES = 50 * 1024 * 1024
c = Counter()
name_hits = []
full_hits = []
num_only = 0
big_files = []
for root in ROOTS:
    for dirpath, dirnames, filenames in os.walk(root, followlinks=False):
        if "934607" in dirpath:
            name_hits.append(("dir", dirpath))
        for fn in filenames:
            p = os.path.join(dirpath, fn)
            c["entries"] += 1
            if "934607" in fn:
                name_hits.append(("file", p))
            try:
                st = os.lstat(p)
            except OSError:
                c["lstat_failed"] += 1
                continue
            if os.path.islink(p):
                c["file_symlink_skipped"] += 1
                continue
            if not os.path.isfile(p):
                c["not_regular_file"] += 1
                continue
            if st.st_size > MAX_BYTES:
                c["over_50MiB_skipped"] += 1
                big_files.append((st.st_size, p))
                continue
            c["read_attempted"] += 1
            try:
                with open(p, "rb") as fh:
                    data = fh.read()
            except OSError:
                c["read_failed"] += 1
                continue
            c["read_ok"] += 1
            if b"934607.nqsv" in data:
                full_hits.append(p)
            elif b"934607" in data:
                num_only += 1
print("# counters (no extension filter)")
for k in sorted(c):
    print(f"{k}\t{c[k]}")
print(f"# path-name hits for 934607: {len(name_hits)}")
for kind, p in name_hits:
    print(f"{kind}\t{p}")
print(f"# files (<=50MiB, any extension) containing '934607.nqsv': {len(full_hits)}")
for p in sorted(full_hits):
    print(p)
print(f"# files (<=50MiB, any extension) containing only digits 934607: {num_only}")
print("# over-50MiB files: name pattern summary")
import re
pat = Counter(re.sub(r"\d+", "#", os.path.basename(p)) for _, p in big_files)
for k, v in pat.most_common():
    print(f"{v}\t{k}")
print(f"# over-50MiB total bytes: {sum(s for s, _ in big_files)}")
print(f"# smallest over-50MiB: {min(s for s, _ in big_files) if big_files else 0}")
