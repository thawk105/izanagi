#!/usr/bin/env python3
"""T-1878: 名指し job dir / evidence dir で PBS request 934607 の痕跡と verifier 出力 file を探す (read-only)."""
import os
import sys

ROOTS = [
    "/work/1/SFC/tanab/dev-wave-jobs/t755-mocc-trace-execution",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-mocc-trace-execution",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-q2-mocc-trace",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t755-q2-mocc-trace-continuation",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1641-t755-report-binding",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1506-mocc-trace0",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t1582-mocc-trace0-pilot",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-mocc-trace-pair",
    "/work/1/SFC/tanab/dev-wave-jobs/dev-wave-mocc-g2-repro-20260826",
    "/work/1/SFC/tanab/izanagi-job-evidence",
]
NEEDLE = b"934607"
TEXT_EXT = {".json", ".md", ".log", ".txt", ".jsonl", ".sha256", ".journal", ".stdout", ".stderr", ".sh", ".rc", ".pid", ".done", ".yaml", ".yml", ".csv", ".tsv", ""}
MAX_BYTES = 50 * 1024 * 1024

hits = []
named = []
scanned = 0
skipped_big = 0
for root in ROOTS:
    if not os.path.isdir(root):
        print(f"MISSING_DIR {root}")
        continue
    for dirpath, dirnames, filenames in os.walk(root):
        for fn in filenames:
            p = os.path.join(dirpath, fn)
            low = fn.lower()
            if low in ("verifier.json", "mocc-trace-pilot-receipt.json", "trace-manifest.json", "mocc-trace-pilot-receipt.sha256") or low.startswith("verifier"):
                named.append(p)
            ext = os.path.splitext(fn)[1].lower()
            if ext not in TEXT_EXT and not fn.endswith(".nqsv"):
                continue
            try:
                st = os.lstat(p)
            except OSError:
                continue
            if not os.path.isfile(p) or os.path.islink(p):
                continue
            if st.st_size > MAX_BYTES:
                skipped_big += 1
                continue
            scanned += 1
            try:
                with open(p, "rb") as fh:
                    data = fh.read()
            except OSError:
                continue
            if NEEDLE in data:
                # 数値の偶然一致を除くため、直後が '.nqsv' か行末/空白/引用符かを示す
                idx = data.find(NEEDLE)
                ctx = data[max(0, idx - 40): idx + 60].replace(b"\n", b"\\n")
                hits.append((p, ctx.decode("utf-8", "replace")))

print(f"scanned_text_files={scanned} skipped_big={skipped_big}")
print(f"--- hits for {NEEDLE.decode()} ({len(hits)}) ---")
for p, ctx in hits:
    print(f"{p}\n    ctx: {ctx}")
print(f"--- named verifier/receipt files ({len(named)}) ---")
for p in named:
    print(p)
