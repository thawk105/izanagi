#!/usr/bin/env bash
# T-2243: run only inside the single allocated compute job.
set -u
export PYTHONDONTWRITEBYTECODE=1 IZANAGI_TASK_RUN_AUTO_RECORD=0
unset PYTEST_ADDOPTS PYTEST_XDIST_TESTRUNUID IZANAGI_TASK_RUN_SIDECAR PYTHONPYCACHEPREFIX
export LC_ALL=C
WAVE_ROOT="$(pwd -P)"
OUT_ROOT=${T2243_OUT:-/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2243-collection-diag/job-out}
USER=$(id -un)
if [[ -d /scr && -w /scr ]]; then LOCAL=/scr/$USER/t2243-$$; else LOCAL=/tmp/t2243-$$; fi
# Exclusively own this scratch directory. Never reuse an earlier result directory.
mkdir -p "$(dirname "$LOCAL")" || exit 2
mkdir "$LOCAL" || exit 2
FINAL_RC=0
stop_tree() {
    local parent=$1 child
    while read -r child; do
        [[ -n $child ]] || continue
        stop_tree "$child"
        kill -TERM "$child" 2>/dev/null || :
    done < <(awk -v p="$parent" '$2 == p {print $1}' <<< "$PROCESS_TREE")
}
stop_children() {
    local PROCESS_TREE
    PROCESS_TREE=$(ps -eo pid=,ppid=)
    stop_tree "$1"
}
finish() {
    local rc=$?
    trap - EXIT INT TERM HUP
    if (( rc != 0 )); then FINAL_RC=$rc; stop_children "$$"; wait 2>/dev/null || :; fi
    if [[ -d $LOCAL/out ]]; then
        rsync -a "$LOCAL/out/" "$OUT_ROOT/" || FINAL_RC=2
    fi
    rm -rf -- "$LOCAL" || FINAL_RC=2
    # This is the only write of the terminal marker, after copying and cleanup.
    printf 'rc=%s\n' "$FINAL_RC" > "$OUT_ROOT/probe.done" || FINAL_RC=2
    exit "$FINAL_RC"
}
trap finish EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
trap 'exit 129' HUP
mkdir -p "$OUT_ROOT" "$LOCAL/out" "$LOCAL/tmp" || exit 2
OUT_ROOT=$(cd "$OUT_ROOT" && pwd -P) || exit 2
export TMPDIR=$LOCAL/tmp
O=$LOCAL/out
elapsed() { awk -v a="$1" -v b="$2" 'BEGIN {printf "%.9f\n", b-a}'; }
lustre() {
    local f seen=0 failed=0
    for f in /proc/fs/lustre/llite/*/stats /proc/fs/lustre/mdc/*/md_stats; do
        [[ -e $f ]] || continue
        seen=1
        printf '### %s\n' "$f"
        cat "$f" || failed=1
    done
    if (( !seen || failed )); then echo 'lustre_stats=unreadable'; return 1; fi
}
singleton() {
    local dir=$1
    ps -eo user,pid,comm --no-headers > "$dir/singleton-before.txt"
    # ps may truncate long user names; compare both spellings of our uid.
    local short_user
    short_user=$(ps -o user= -p $$ | xargs)
    awk -v me="$USER" -v short="$short_user" '
      $1 != me && $1 != short && $1 !~ /^(root|daemon|nqs.*|message.*|systemd.*|syslog|_.*|bin|sys|sync|games|man|lp|mail|news|uucp|proxy|www-data|backup|list|irc|gnats|nobody|sshd|ntp|chrony|rpc|rpcuser|dbus|polkitd|postfix|munge)$/ {n++}
      END {print "others=" n+0}' "$dir/singleton-before.txt"
    ps -eo uid=,pid=,comm=,args= > "$dir/process-args-before.txt"
    awk -v uid="$(id -u)" '$1 == uid && ($3 ~ /^pytest/ || ($3 ~ /^python/ && $0 ~ / -m pytest( |$)/)) {n++}
      END {print "stale_pytest=" n+0}' "$dir/process-args-before.txt"
}
wave_cache() {
    local phase=$1 dirs pycs
    dirs=$(find "$WAVE_ROOT/orchestrator" "$WAVE_ROOT/tools" -name '__pycache__' -type d | wc -l)
    pycs=$(find "$WAVE_ROOT/orchestrator" "$WAVE_ROOT/tools" -name '*.pyc' | wc -l)
    printf 'pycache_dirs=%s\npyc=%s\n' "$dirs" "$pycs" > "$O/pycache-wave-$phase.txt"
    if (( dirs || pycs )); then echo 'R_cache=warm'; else echo 'R_cache=cold'; fi >> "$O/pycache-wave-$phase.txt"
}
PERF=
P=/usr/lib/linux-tools/5.15.0-135-generic/perf
if [[ -x $P ]] && "$P" stat -e cycles -- true > "$O/perf-check.txt" 2>&1; then PERF=$P; fi
LUSTRE=readable
lustre > "$O/lustre-initial.txt" 2>&1 || LUSTRE=unreadable
{
    hostname; date --iso-8601=seconds; uname -r; nproc; free -g
    df -h /scr /tmp "$WAVE_ROOT"
    mount | grep -E ' /scr | /tmp | /work '
    python3 --version; python3 -m pytest --version
    printf 'perf=%s\nlustre_stats=%s\nWAVE_ROOT=%s\n' "${PERF:-absent}" "$LUSTRE" "$WAVE_ROOT"
    git -C "$WAVE_ROOT" rev-parse HEAD
    git -C "$WAVE_ROOT" status --porcelain | wc -l
    singleton "$O"
} > "$O/env.txt" 2>&1
wave_cache before
find "$WAVE_ROOT/orchestrator" "$WAVE_ROOT/tools" -name '*.py' -exec cat {} + > /dev/null
C_VALID=1
run_cell() {
    local name=$1 arm=$2 n=$3 half=$4 warm=${5:-0}
    local dir=$O/$name cwd=$WAVE_ROOT prefix= start end i rc sampler
    local -a pids=() rcs=() wrapper=() envcmd=(env) args=(orchestrator/tests --collect-only -q -p no:cacheprovider)
    [[ $arm == L || $arm == X ]] && prefix=$OUT_ROOT/pycache-L
    if [[ $arm == C ]]; then cwd=$LOCAL/repo; prefix=$LOCAL/pycache-C; fi
    mkdir -p "$dir" || exit 2
    {
        printf 'arm=%s\nN=%s\nhalf=%s\ncwd=%s\nprefix=%s\n' "$arm" "$n" "$half" "$cwd" "$prefix"
        singleton "$dir"
        nproc | sed 's/^/nproc=/'
        taskset -pc $$
    } > "$dir/cell.txt" 2>&1
    cat /proc/loadavg > "$dir/loadavg-before.txt"
    grep '^MemAvailable:' /proc/meminfo > "$dir/mem-before.txt"
    lustre > "$dir/lustre-before.txt" 2>&1 || echo 'lustre_stats=unreadable' >> "$dir/cell.txt"
    if [[ $arm == C && $C_VALID == 0 ]]; then
        echo 'skipped=set-mismatch' >> "$dir/cell.txt"
        return
    fi
    [[ -n $PERF ]] && wrapper=("$PERF" stat -x, -e cycles,instructions,cache-misses)
    if [[ $warm == 1 ]]; then envcmd+=(-u PYTHONDONTWRITEBYTECODE); fi
    if [[ -n $prefix ]]; then envcmd+=("PYTHONPYCACHEPREFIX=$prefix"); else envcmd+=(-u PYTHONPYCACHEPREFIX); fi
    local count=$n
    if [[ $arm == X ]]; then count=1; args=(orchestrator/tests -n 48 -k zzz_no_such_test_zzz -q -p no:cacheprovider); fi
    start=$(date +%s.%N)
    (sleep 10; grep MHz /proc/cpuinfo | head -8 > "$dir/cpufreq.txt") &
    sampler=$!
    for ((i=0; i<count; i++)); do
        (
            cd "$cwd" || exit 2
            if [[ -n $PERF ]]; then
                exec "${envcmd[@]}" "${wrapper[@]}" -o "$dir/proc-$i.perf" -- /usr/bin/time -f '%e %U %S %M %F %R %w %c' -o "$dir/proc-$i.time" python3 -m pytest "${args[@]}"
            else
                exec "${envcmd[@]}" /usr/bin/time -f '%e %U %S %M %F %R %w %c' -o "$dir/proc-$i.time" python3 -m pytest "${args[@]}"
            fi
        ) > "$dir/proc-$i.out" 2> "$dir/proc-$i.err" &
        pids+=("$!")
    done
    for ((i=0; i<count; i++)); do
        wait "${pids[$i]}"; rc=$?
        printf '%s\n' "$rc" >> "$dir/proc-$i.rc"
        rcs+=("$rc")
    done
    end=$(date +%s.%N)
    stop_children "$sampler"
    kill "$sampler" 2>/dev/null || :
    wait "$sampler" 2>/dev/null || :
    {
        printf 'cohort_start=%s\ncohort_end=%s\ncohort_wall=%s\n' "$start" "$end" "$(elapsed "$start" "$end")"
        printf 'rc_list=%s\n' "${rcs[*]}"
        printf 'expected_rc=%s\n' "$([[ $arm == X ]] && echo 5 || echo 0)"
    } >> "$dir/cell.txt"
    lustre > "$dir/lustre-after.txt" 2>&1 || echo 'lustre_stats=unreadable' >> "$dir/cell.txt"
    cat /proc/loadavg > "$dir/loadavg-after.txt"
    grep '^MemAvailable:' /proc/meminfo > "$dir/mem-after.txt"
}
run_cell R1-a R 1 a
run_cell R48-a R 48 a
run_cell warm-L L 1 warm 1
[[ $(cat "$O/warm-L/proc-0.rc") == 0 ]] || exit 3
{
    printf 'pyc='; find "$OUT_ROOT/pycache-L" -name '*.pyc' | wc -l
    printf 'pytest_pyc='; find "$OUT_ROOT/pycache-L" -name '*pytest*.pyc' | wc -l
    printf 'rewrite_pyc='; find "$OUT_ROOT/pycache-L" -name '*-pytest-*.pyc' | wc -l
} > "$O/pycache-L-count.txt"
start=$(date +%s.%N)
git clone --depth 1 --no-local "file://$WAVE_ROOT" "$LOCAL/repo" > "$O/clone.out" 2> "$O/clone.err"
rc=$?; end=$(date +%s.%N)
printf 'clone_rc=%s\nclone_seconds=%s\n' "$rc" "$(elapsed "$start" "$end")" > "$O/stage-C.txt"
(( rc == 0 )) || exit 4
start=$(date +%s.%N)
rsync -a --exclude .git "$WAVE_ROOT/external/" "$LOCAL/repo/external/" > "$O/rsync.out" 2> "$O/rsync.err"
rc=$?; end=$(date +%s.%N)
printf 'rsync_rc=%s\nrsync_seconds=%s\n' "$rc" "$(elapsed "$start" "$end")" >> "$O/stage-C.txt"
(( rc == 0 )) || C_VALID=0
du -sm "$LOCAL/repo" >> "$O/stage-C.txt"
# Still attempt warm-C after a staging error to retain diagnostic evidence.
saved_valid=$C_VALID; C_VALID=1
run_cell warm-C C 1 warm 1
C_VALID=$saved_valid
wave_head=$(git -C "$WAVE_ROOT" rev-parse HEAD)
local_head=$(git -C "$LOCAL/repo" rev-parse HEAD)
head_match=0
[[ -n $wave_head && $wave_head == "$local_head" ]] && head_match=1
printf 'wave_head=%s\nlocal_head=%s\nhead_match=%s\n' "$wave_head" "$local_head" "$head_match" >> "$O/stage-C.txt"
(( head_match )) || C_VALID=0
grep '::' "$O/warm-L/proc-0.out" | sort > "$O/set-L.txt"
grep '::' "$O/warm-C/proc-0.out" | sort > "$O/set-C.txt"
diff -u "$O/set-L.txt" "$O/set-C.txt" > "$O/set.diff"; diff_rc=$?
l_rc=$(cat "$O/warm-L/proc-0.rc"); c_rc=$(cat "$O/warm-C/proc-0.rc")
[[ $diff_rc == 0 && $l_rc == "$c_rc" ]] || C_VALID=0
{
    wc -l "$O/set-L.txt" "$O/set-C.txt" "$O/set.diff"
    printf 'diff_rc=%s\nL_rc=%s\nC_rc=%s\nC_VALID=%s\n' "$diff_rc" "$l_rc" "$c_rc" "$C_VALID"
    head -20 "$O/set.diff"
} > "$O/setcheck.txt"
run_cell ref-L48-pre L 48 pre
for n in 1 4 12 24 48; do run_cell "L$n-a" L "$n" a; run_cell "C$n-a" C "$n" a; done
run_cell X48-a X 48 a
for n in 48 24 12 4 1; do run_cell "C$n-b" C "$n" b; run_cell "L$n-b" L "$n" b; done
run_cell X48-b X 48 b
run_cell R48-b R 48 b
run_cell R1-b R 1 b
run_cell ref-L48-post L 48 post
wave_cache after
exit 0
