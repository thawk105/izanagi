#!/bin/bash
# [T-2243] 診断 job の投入 launcher (親の運転 script、repo 外)。
# 計算ノード 1 job (generic) で wave 専用 dir に複製した probe を走らせる。cwd は dispatcher が wave 木に固定する。
D=/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2243-collection-diag
echo $$ > "$D/run-probe-1.pid"
# login 側の dispatcher が wave 木に pyc を書いて R 腕 (冷) を汚さないようにする
export PYTHONDONTWRITEBYTECODE=1
cd /work/1/SFC/tanab/izanagi/.claude/worktrees/dev-wave-t2243-collection-diag || { echo "rc=3" > "$D/run-probe-1.done"; exit 3; }
date '+%Y-%m-%dT%H:%M:%S%z start' >> "$D/run-probe-1.times"
git rev-parse HEAD > "$D/run-probe-1.tip.txt"
python3 tools/pegasus/dispatch_compute.py --task generic \
  --walltime 01:00:00 --queue-wait-timeout 5400 --overall-grace 5400 \
  -- bash "$D/t2243_collection_contention_probe.sh" > "$D/run-probe-1.log" 2>&1
rc=$?
date "+%Y-%m-%dT%H:%M:%S%z end rc=$rc" >> "$D/run-probe-1.times"
echo "rc=$rc" > "$D/run-probe-1.done"
exit "$rc"
