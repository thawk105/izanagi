"""Compute-node liveness check for [T-2320] (repo 外、使い捨て、commit しない)。v2 (段 6 レビュー B の must-fix 3 件を反映)。

09-04 の login-node 再現 (第 2 層を確定した形) を、修正後の経路で計算ノードで走らせる:
  prepare_masstree_fetchcontent (patched scratch root, canonical temp base)
  -> capture_define_inputs(..., configure_args=("-DFETCHCONTENT_BASE_DIR=<base>",))
  -> 両 driver の unique 値集合 (A-5 / T-2266) について supply arm を 1 件ずつ発行して record を保存
  -> 修正後 helper `_require_backoff_condition_gate` を同じ入力で driver ごとに 1 回呼び admission を保存

機械検査:
  layer2_gone (必要条件): record 件数 = 14、preprocess-failed 0 件、各 record の requested/control
    configure argv に `-DFETCHCONTENT_BASE_DIR=` が exact 1 個ずつあり、その値が prepare の base と一致。
  job_ready (十分条件): layer2_gone かつ 14 record すべて green かつ両 helper run が admitted。
rc 0 は job_ready のときだけ。rc 3 はそれ以外 (JSON に理由が残る)。
これは測定ではなく、job の記録でもない。record は一時 base path を含む再生不能な実行時観測。
"""
import glob
import hashlib
import json
import os
import subprocess
import sys
import tempfile
import time
import traceback

REPO = sys.argv[1]
OUT_DIR = sys.argv[2]
DEPS = "/work/1/SFC/tanab/izanagi-a2-deps"
# job body (a5_second_boot_backoff_sweep.sh:32, b10_backoff_grid.sh:23) と同じ proxy。
BUILD_NETWORK_PROXY_URL = "http://10.120.96.1:8080"

os.makedirs(OUT_DIR, exist_ok=True)
SCRATCH = os.path.join(OUT_DIR, "scratch")
os.makedirs(SCRATCH, exist_ok=True)
os.environ["TMPDIR"] = SCRATCH
os.environ["CMAKE_PREFIX_PATH"] = DEPS
os.environ["http_proxy"] = BUILD_NETWORK_PROXY_URL
os.environ["https_proxy"] = BUILD_NETWORK_PROXY_URL
sys.path.insert(0, REPO)

from orchestrator.campaign import backoff_sweep, buildcache, p2_2, patchharness, pin, site_policy  # noqa: E402
from orchestrator.campaign import condition_meaning_gate as cmg  # noqa: E402

A5_DRIVER = "orchestrator/campaign/backoff_sweep.py"
T2266_DRIVER = "orchestrator/campaign/backoff_extended_sweep.py"
A5_VALUES = (-1, 2, 5, 10, 25, 50, 100)
T2266_VALUES = (-1, 150, 200, 300, 500, 750, 999)
EXPECTED_RECORDS = len(A5_VALUES) + len(T2266_VALUES)

ccbench = os.path.join(REPO, "external", "ccbench")
patch = os.path.join(REPO, "patches", "silo-backoff-fixed.patch")
cc, cxx = buildcache.compilers_for_current_site()
manifest = buildcache.observed_toolchain_manifest(cc, cxx)
site_policy_site = site_policy.current_site()
# production 2 経路と同じ入口 (p2_2.resolve_site_runtime) で site を取る。
runtime_site, _contract, _authorization = p2_2.resolve_site_runtime()
site = runtime_site

result = {
    "note": "T-2320 compute-node liveness check v2; not a measurement, not a job record; "
            "records contain a temporary base path and are non-replayable observations",
    "repo": REPO, "repo_head": None, "ccbench_pin": pin.CURRENT_PIN,
    "site": site, "site_policy_site": site_policy_site,
    "site_matches_site_policy": site == site_policy_site,
    "cc": cc, "cxx": cxx, "hostname": os.uname().nodename,
    "cmake_prefix_path": os.environ["CMAKE_PREFIX_PATH"],
    "proxy": BUILD_NETWORK_PROXY_URL,
    "prepare": None, "config_h_after_prepare": [], "config_h_after_gate": [],
    "supply_records": [], "helper_runs": [], "error": None,
}
try:
    result["repo_head"] = subprocess.run(
        ["git", "-C", REPO, "rev-parse", "HEAD"], capture_output=True, text=True, check=True,
    ).stdout.strip()
except Exception as exc:  # noqa: BLE001
    result["repo_head"] = f"unavailable: {exc}"


def sha256_file(path):
    with open(path, "rb") as handle:
        return hashlib.sha256(handle.read()).hexdigest()


def masstree_config_h(base):
    out = []
    for path in sorted(glob.glob(os.path.join(base, "**", "config.h"), recursive=True)):
        if "masstree" in os.path.relpath(path, base).lower():
            out.append({"path": path, "sha256": sha256_file(path)})
    return out


def base_args(argv):
    return [a for a in argv if a.startswith("-DFETCHCONTENT_BASE_DIR=")]


def record_dict(value, record):
    requested_argv = list(record.evidence.get("requested_configure_argv", ()))
    control_argv = list(record.evidence.get("control_configure_argv", ()))
    return {
        "requested_value": value,
        "record_driver_id": getattr(record, "driver_id", None),
        "arm": record.arm,
        "macro": record.macro,
        "terminal_status": record.terminal_status,
        "reason_code": record.reason_code,
        "requested_configure_argv": requested_argv,
        "control_configure_argv": control_argv,
        "requested_base_args": base_args(requested_argv),
        "control_base_args": base_args(control_argv),
        "detail": record.evidence.get("detail"),
    }


prepare_base = None
try:
    with patchharness.checkout(pin.CURRENT_PIN, base_dir=ccbench) as stock_root:
        with patchharness.checkout(pin.CURRENT_PIN, base_dir=ccbench) as variant_root:
            with patchharness.applied(patch, pin.CURRENT_PIN, ccbench_dir=variant_root):
                with tempfile.TemporaryDirectory(
                        prefix="izanagi-backoff-condition-gate-", dir=SCRATCH) as base_raw:
                    base = os.path.realpath(base_raw)
                    prepare_base = base
                    root = os.path.realpath(variant_root)
                    t0 = time.time()
                    prep = buildcache.prepare_masstree_fetchcontent(
                        ccbench_dir=root, fetchcontent_base_dir=base,
                        expected_toolchain_manifest=manifest,
                        configure_timeout_s=900, target_timeout_s=900, site=site,
                    )
                    result["prepare"] = {
                        "elapsed_s": round(time.time() - t0, 1),
                        "base": base, "source_root": root, "stock_root": os.path.realpath(stock_root),
                        "result_repr": repr(prep)[:2000],
                    }
                    result["config_h_after_prepare"] = masstree_config_h(base)
                    configure_args = (f"-DFETCHCONTENT_BASE_DIR={base}",)
                    captured = cmg.capture_define_inputs(
                        root, stock_root=stock_root, configure_args=configure_args,
                    )
                    for driver_id, values in ((A5_DRIVER, A5_VALUES), (T2266_DRIVER, T2266_VALUES)):
                        for value in values:
                            stock_comparison = value == -1
                            request = cmg.make_define_request(
                                driver_id=driver_id, macro="BACKOFF_FIXED",
                                requested_value=value,
                                default_value=None if stock_comparison else -1,
                                stock_comparison=stock_comparison,
                            )
                            t1 = time.time()
                            record = cmg.evaluate_define_supply_effectuation(
                                captured, request=request, cxx=cxx, cmake="cmake",
                            )
                            entry = record_dict(value, record)
                            entry["loop_driver_id"] = driver_id
                            entry["elapsed_s"] = round(time.time() - t1, 1)
                            result["supply_records"].append(entry)
                            print(f"{driver_id} BACKOFF_FIXED={value}: "
                                  f"{record.terminal_status}/{record.reason_code}", flush=True)
                    result["config_h_after_gate"] = masstree_config_h(base)
                    for driver_id, values in ((A5_DRIVER, A5_VALUES), (T2266_DRIVER, T2266_VALUES)):
                        entry = {"driver_id": driver_id, "values": list(values)}
                        t2 = time.time()
                        try:
                            run = backoff_sweep._require_backoff_condition_gate(
                                root, stock_root=stock_root, driver_id=driver_id,
                                macro_values={"BACKOFF_FIXED": values}, cxx=cxx,
                                configure_args=configure_args,
                            )
                            entry["admitted"] = bool(run.admission.admitted)
                            entry["supply"] = [
                                (r.macro, r.terminal_status, r.reason_code) for r in run.supply_records
                            ]
                            entry["meaning"] = [
                                (r.macro, r.terminal_status, r.reason_code) for r in run.meaning_records
                            ]
                        except Exception as exc:  # noqa: BLE001
                            entry["admitted"] = False
                            entry["error"] = f"{type(exc).__name__}: {exc}"[:4000]
                        entry["elapsed_s"] = round(time.time() - t2, 1)
                        result["helper_runs"].append(entry)
                        print(f"helper {driver_id}: admitted={entry['admitted']}", flush=True)
except Exception as exc:  # noqa: BLE001
    result["error"] = f"{type(exc).__name__}: {exc}\n{traceback.format_exc()}"[:8000]
    print("ERROR", result["error"], flush=True)

records = result["supply_records"]
expected_base_arg = f"-DFETCHCONTENT_BASE_DIR={prepare_base}" if prepare_base else None
check = {
    "record_count": len(records),
    "expected_record_count": EXPECTED_RECORDS,
    "preprocess_failed_count": sum(1 for r in records if r["reason_code"] == "preprocess-failed"),
    "exact_base_match_count": sum(
        1 for r in records
        if expected_base_arg is not None
        and r["requested_base_args"] == [expected_base_arg]
        and r["control_base_args"] == [expected_base_arg]
    ),
    "green_count": sum(1 for r in records if r["terminal_status"] == "green"),
    "driver_id_match_count": sum(
        1 for r in records if r["record_driver_id"] == r["loop_driver_id"]
    ),
    "helper_admitted": [bool(h.get("admitted")) for h in result["helper_runs"]],
    "config_h_stable_across_gate": (
        result["config_h_after_prepare"] == result["config_h_after_gate"]
        and len(result["config_h_after_prepare"]) >= 1
    ),
    "top_level_error": result["error"] is not None,
}
check["layer2_gone"] = (
    check["record_count"] == EXPECTED_RECORDS
    and check["preprocess_failed_count"] == 0
    and check["exact_base_match_count"] == EXPECTED_RECORDS
)
check["job_ready"] = (
    check["layer2_gone"]
    and check["green_count"] == EXPECTED_RECORDS
    and len(check["helper_admitted"]) == 2
    and all(check["helper_admitted"])
    and not check["top_level_error"]
)
result["check"] = check
out = os.path.join(OUT_DIR, "liveness.json")
with open(out, "w", encoding="utf-8") as handle:
    json.dump(result, handle, indent=1, default=str, ensure_ascii=False)
print("wrote", out, "check=", json.dumps(check), flush=True)
sys.exit(0 if check["job_ready"] else 3)
