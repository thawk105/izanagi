#!/usr/bin/env python3
"""per-page=1 の 1 credit で今窓の残量を観測する (repo 外の使い捨て probe)。"""
import datetime
import json
import urllib.error
import urllib.request

req = urllib.request.Request(
    "https://api.openalex.org/works?per-page=1",
    headers={"Accept": "application/json"},
)
try:
    handle = urllib.request.urlopen(req, timeout=60)
    status, hdrs = handle.status, handle.headers
except urllib.error.HTTPError as exc:
    status, hdrs = exc.code, exc.headers
sel = {k.lower(): v for k, v in hdrs.items() if k.lower().startswith("x-ratelimit")}
now = datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat()
print(status, json.dumps(sel, sort_keys=True), now)
