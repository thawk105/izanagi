# -*- coding: utf-8 -*-
"""T-673 D1 走査完全性 guard の事前登録済み overhead 測定器。"""
from __future__ import annotations

import json
import math
import os
import platform
import random
import socket

import subprocess
import sys
import time
from collections.abc import Callable, Mapping, Sequence
from pathlib import Path

from typing import Any

import pytest


M_VALUES = (2, 8, 64)
ORDER_SEED = 20260810
BOOTSTRAP_SEED = 20260811
BOOTSTRAP_RESAMPLES = 10_000
TARGET_BLOCK_NS = 500_000_000
BLOCK_PAIR_COUNT = 32
SCHEDULE_COUNT = BLOCK_PAIR_COUNT // 2
RUNTIME_LIMIT_SECONDS = 600.0
RUNTIME_RESERVE_SECONDS = 30.0
WARMUP_CALLS_PER_ARM = 3


def _fail(message: str) -> None:
    pytest.fail(message, pytrace=False)


def _required_env_path(name: str, *, python_source: bool) -> Path:
    raw = os.environ.get(name)
    if raw is None or not raw.strip():
        _fail(f"必須環境変数 {name} が未設定である")
    path = Path(raw).expanduser().resolve()
    if python_source and path.suffix != ".py":
        _fail(f"{name} は .py ファイルを指していない: {path}")
    if python_source and not path.is_file():
        _fail(f"{name} が指す Python ファイルを読めない: {path}")
    return path


def _load_module(path: Path, module_name: str) -> ModuleType:
    if module_name in sys.modules:
        _fail(f"測定用 module 名が既に使われている: {module_name}")
    spec = importlib.util.spec_from_file_location(module_name, path)
    if spec is None or spec.loader is None:
        _fail(f"module spec を作れない: {path}")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    old_dont_write_bytecode = sys.dont_write_bytecode
    sys.dont_write_bytecode = True
    try:
        spec.loader.exec_module(module)
    except BaseException:
        sys.modules.pop(module_name, None)
        raise
    finally:
        sys.dont_write_bytecode = old_dont_write_bytecode
    return module


def _sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as source:
        for chunk in iter(lambda: source.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def _git_head(path: Path) -> str:
    environment = dict(os.environ)
    environment["GIT_OPTIONAL_LOCKS"] = "0"
    try:
        completed = subprocess.run(
            ["git", "-C", str(path.parent), "rev-parse", "--verify", "HEAD"],
            check=True,
            capture_output=True,
            text=True,
            timeout=10,
            env=environment,
        )
    except (OSError, subprocess.SubprocessError) as exc:
        _fail(f"commit OID を取得できない ({path}): {exc}")
    oid = completed.stdout.strip()
    if len(oid) != 40 or any(character not in "0123456789abcdef" for character in oid):
        _fail(f"commit OID が 40 桁 lower-hex でない: {oid!r}")
    return oid


def _contract_hash(env_tag: str, generation: int) -> str:
    material = f"{env_tag}\0generation={generation}".encode("ascii")
    return hashlib.sha256(material).hexdigest()


def _catalog(env_count: int) -> dict[str, tuple[tuple[int, str], ...]]:
    return {
        env_tag: tuple(
            (generation, _contract_hash(env_tag, generation))
            for generation in (1, 2)
        )
        for env_tag in (f"env-{index:03d}" for index in range(env_count))
    }


def _raw(module: ModuleType, document: Mapping[str, object]) -> bytes:
    return module.canonical_record_bytes(document) + b"\n"


def _build_case(module: ModuleType, env_count: int) -> Callable[[], object]:
    """module 固有の ActiveContract だけで、受理される二 record chain を作る。"""
    catalog = _catalog(env_count)
    env_tags = sorted(catalog)
    records: list[tuple[str, bytes]] = []
    previous_hash: str | None = None
    head: dict[str, object] | None = None
    for serial, generation in enumerate((1, 2), start=1):
        rows = tuple(
            module.ActiveContract(
                env_tag=env_tag,
                generation=generation,
                contract_sha256=dict(catalog[env_tag])[generation],
            )
            for env_tag in env_tags
        )
        head = module.build_activation_record(
            activation_serial=serial,
            previous_activation_state_sha256=previous_hash,
            active_contracts=rows,
        )
        records.append((f"{serial:08d}.json", _raw(module, head)))
        previous_hash = head["activation_state_sha256"]
    assert head is not None
    frozen_records = tuple(records)
    expected_serial = head["activation_serial"]
    expected_hash = head["activation_state_sha256"]

    def predicate(_predecessor: object, _successor: object) -> bool:
        return True

    def invoke() -> object:
        return module.validate_activation_records(
            frozen_records,
            registered_contracts=catalog,
            is_valid_registered_successor=predicate,
            expected_head_serial=expected_serial,
            expected_head_state_sha256=expected_hash,
        )

    return invoke


def _empty_callable() -> None:
    return None


def _time_block(call: Callable[[], object], k: int) -> int:
    started_ns = time.perf_counter_ns()
    for _ in range(k):
        call()
    return time.perf_counter_ns() - started_ns


def _block(arm: str, raw_total_ns: int, k: int, sequence_index: int) -> dict[str, object]:
    return {
        "arm": arm,
        "sequence_index": sequence_index,
        "k": k,
        "raw_total_ns": raw_total_ns,
        "ns_per_call": raw_total_ns / k,
    }


def _calibrate(
    call_a: Callable[[], object],
    call_b: Callable[[], object],
) -> tuple[int, list[dict[str, object]]]:
    k = 1
    attempts: list[dict[str, object]] = []
    while True:
        raw_a = _time_block(call_a, k)
        raw_b = _time_block(call_b, k)
        attempts.append({
            "k": k,
            "order": "A_then_B",
            "A": {
                "raw_total_ns": raw_a,
                "ns_per_call": raw_a / k,
            },
            "B": {
                "raw_total_ns": raw_b,
                "ns_per_call": raw_b / k,
            },
            "discarded_from_statistics": True,
        })
        if raw_a >= TARGET_BLOCK_NS and raw_b >= TARGET_BLOCK_NS:
            return k, attempts
        k *= 2


def _percentile(values: Sequence[float], probability: float) -> float:
    if not values:
        raise ValueError("percentile には非空列が必要")
    ordered = sorted(values)
    position = (len(ordered) - 1) * probability
    lower = math.floor(position)
    upper = math.ceil(position)
    if lower == upper:
        return float(ordered[lower])
    weight = position - lower
    return float(ordered[lower] * (1.0 - weight) + ordered[upper] * weight)


def _bootstrap_median_ci(
    differences: Sequence[float],
    rng: random.Random,
) -> tuple[float, float]:
    sample_size = len(differences)
    medians = [
        float(statistics.median(
            differences[rng.randrange(sample_size)] for _ in range(sample_size)
        ))
        for _ in range(BOOTSTRAP_RESAMPLES)
    ]
    return _percentile(medians, 0.025), _percentile(medians, 0.975)


def _same_sign_fraction(differences: Sequence[float], median: float) -> float:
    if median > 0:
        return sum(value > 0 for value in differences) / len(differences)
    if median < 0:
        return sum(value < 0 for value in differences) / len(differences)
    return 0.0


def _statistics(
    paired_differences: Sequence[float],
    same_arm_differences: Sequence[float],
    empty_ns_per_call: Sequence[float],
    bootstrap_rng: random.Random,
    *,
    protocol_complete: bool,
) -> dict[str, object]:
    median = float(statistics.median(paired_differences))
    ci_low, ci_high = _bootstrap_median_ci(paired_differences, bootstrap_rng)
    same_sign = _same_sign_fraction(paired_differences, median)
    same_arm_p95 = _percentile(
        [abs(value) for value in same_arm_differences], 0.95
    )
    empty_p95 = _percentile(empty_ns_per_call, 0.95)
    conditions = {
        "ci_excludes_zero": ci_low > 0.0 or ci_high < 0.0,
        "same_sign_at_least_0_8": same_sign >= 0.8,
        "absolute_median_exceeds_same_arm_and_empty_p95": (
            abs(median) > same_arm_p95 and abs(median) > empty_p95
        ),
    }
    missing = [name for name, satisfied in conditions.items() if not satisfied]
    if not protocol_complete:
        missing.append("minimum_block_duration_relaxed_for_runtime_limit")
    ranked = all(conditions.values()) and protocol_complete
    observed_direction = (
        "B_slower_than_A" if median > 0 else
        "B_faster_than_A" if median < 0 else
        "tie"
    )
    return {
        "paired_difference_definition": "B_ns_per_call - A_ns_per_call",
        "paired_difference_median_ns_per_call": median,
        "bootstrap_95_ci_ns_per_call": [ci_low, ci_high],
        "bootstrap_resamples": BOOTSTRAP_RESAMPLES,
        "same_sign_fraction": same_sign,
        "noise_thresholds_ns_per_call": {
            "absolute_A_vs_A_difference_p95": same_arm_p95,
            "empty_callable_p95": empty_p95,
        },
        "ranking_conditions": conditions,
        "ranked": ranked,
        "missing_conditions": missing,
        "observed_median_direction": observed_direction,
        "ranked_direction": observed_direction if ranked else None,
    }


def _maybe_reduce_m64_k(
    m: int,
    k: int,
    calibration: Sequence[Mapping[str, Any]],
    measurement_started: float,
) -> tuple[int, dict[str, object] | None]:
    if m != 64:
        return k, None
    final = calibration[-1]
    raw_a = int(final["A"]["raw_total_ns"])
    raw_b = int(final["B"]["raw_total_ns"])
    # Primary は A/B 各32 block、A-vs-A は A 64 block。空 callable は
    # 十分小さいため 30 秒の reserve 側へ含める。
    projected_seconds = (
        BLOCK_PAIR_COUNT * (raw_a + raw_b) +
        2 * BLOCK_PAIR_COUNT * raw_a
    ) / 1_000_000_000
    elapsed_seconds = time.monotonic() - measurement_started
    available_seconds = (
        RUNTIME_LIMIT_SECONDS - RUNTIME_RESERVE_SECONDS - elapsed_seconds
    )
    if projected_seconds <= available_seconds:
        return k, None
    usable_seconds = max(0.0, available_seconds) * 0.9
    reduced_k = max(1, int(k * usable_seconds / projected_seconds))
    if reduced_k >= k:
        reduced_k = max(1, k - 1)
    return reduced_k, {
        "reason": "10 分上限超過の予測により M=64 の K を低減した",
        "original_k": k,
        "reduced_k": reduced_k,
        "elapsed_before_M64_measurement_seconds": elapsed_seconds,
        "projected_M64_measurement_seconds_at_original_k": projected_seconds,
        "available_seconds_after_30s_reserve": available_seconds,
        "consequence": "0.5 秒 block 条件を満たさない可能性があるため順位を付けない",
    }


def _measure_m(
    m: int,
    call_a: Callable[[], object],
    call_b: Callable[[], object],
    order_rng: random.Random,
    bootstrap_rng: random.Random,
    measurement_started: float,
) -> dict[str, object]:
    for _ in range(WARMUP_CALLS_PER_ARM):
        state_a = call_a()
        state_b = call_b()
    if state_a.activation_serial != 2 or len(state_a.active_contracts) != m:
        _fail(f"A の warm-up が正しい M={m} state を返さなかった")
    if state_b.activation_serial != 2 or len(state_b.active_contracts) != m:
        _fail(f"B の warm-up が正しい M={m} state を返さなかった")

    calibrated_k, calibration = _calibrate(call_a, call_b)
    k, budget_adjustment = _maybe_reduce_m64_k(
        m, calibrated_k, calibration, measurement_started
    )
    protocol_complete = budget_adjustment is None

    schedules = ["ABBA"] * (SCHEDULE_COUNT // 2)
    schedules += ["BAAB"] * (SCHEDULE_COUNT // 2)
    order_rng.shuffle(schedules)

    primary_pairs: list[dict[str, object]] = []
    same_arm_pairs: list[dict[str, object]] = []
    empty_blocks: list[dict[str, object]] = []
    paired_differences: list[float] = []
    same_arm_differences: list[float] = []
    sequence_index = 0
    pair_index = 0
    same_arm_pair_index = 0

    for schedule_index, schedule in enumerate(schedules):
        quartet: list[dict[str, object]] = []
        for arm in schedule:
            raw = _time_block(call_a if arm == "A" else call_b, k)
            quartet.append(_block(arm, raw, k, sequence_index))
            sequence_index += 1
        for offset in (0, 2):
            first, second = quartet[offset:offset + 2]
            by_arm = {str(first["arm"]): first, str(second["arm"]): second}
            difference = float(by_arm["B"]["ns_per_call"]) - float(
                by_arm["A"]["ns_per_call"]
            )
            primary_pairs.append({
                "pair_index": pair_index,
                "schedule_index": schedule_index,
                "schedule": schedule,
                "order": "A_first" if first["arm"] == "A" else "B_first",
                "blocks": [first, second],
                "B_minus_A_ns_per_call": difference,
            })
            paired_differences.append(difference)
            pair_index += 1

        for _ in range(2):
            raw_first = _time_block(call_a, k)
            first = _block("A", raw_first, k, sequence_index)
            sequence_index += 1
            raw_second = _time_block(call_a, k)
            second = _block("A", raw_second, k, sequence_index)
            sequence_index += 1
            difference = float(second["ns_per_call"]) - float(first["ns_per_call"])
            same_arm_pairs.append({
                "pair_index": same_arm_pair_index,
                "order": "A_then_A",
                "blocks": [first, second],
                "second_minus_first_ns_per_call": difference,
            })
            same_arm_differences.append(difference)
            same_arm_pair_index += 1

        for _ in range(2):
            raw_empty = _time_block(_empty_callable, k)
            empty_blocks.append(_block("empty", raw_empty, k, sequence_index))
            sequence_index += 1

    if len(primary_pairs) != BLOCK_PAIR_COUNT:
        _fail("primary block 対数が事前登録値と一致しない")
    if len(same_arm_pairs) != BLOCK_PAIR_COUNT:
        _fail("A-vs-A block 対数が primary と一致しない")
    if len(empty_blocks) != BLOCK_PAIR_COUNT:
        _fail("empty callable block 数が primary block 対数と一致しない")

    stats = _statistics(
        paired_differences,
        same_arm_differences,
        [float(block["ns_per_call"]) for block in empty_blocks],
        bootstrap_rng,
        protocol_complete=protocol_complete,
    )
    return {
        "M": m,
        "K": k,
        "calibrated_K": calibrated_k,
        "target_block_ns": TARGET_BLOCK_NS,
        "warmup_calls_discarded_per_arm": WARMUP_CALLS_PER_ARM,
        "calibration_blocks": calibration,
        "budget_adjustment": budget_adjustment,
        "schedule_counts": {
            "ABBA": schedules.count("ABBA"),
            "BAAB": schedules.count("BAAB"),
        },
        "schedule_order": schedules,
        "block_pair_count": len(primary_pairs),
        "primary_pairs": primary_pairs,
        "noise": {
            "A_vs_A_pairs": same_arm_pairs,
            "empty_callable_blocks": empty_blocks,
        },
        "statistics": stats,
        "operation_conversion": {
            "validate_calls_per_loader_process": 1,
            "validate_calls_per_publish": 1,
            "median_overhead_ns_per_loader_process": stats[
                "paired_difference_median_ns_per_call"
            ],
            "median_overhead_ns_per_publish": stats[
                "paired_difference_median_ns_per_call"
            ],
            "practical_acceptance_threshold": None,
            "threshold_owner": "user",
        },
        "outliers_discarded": 0,
    }


def test_measure_guard_overhead() -> None:
    path_a = _required_env_path("IZ_T673_MOD_A", python_source=True)
    path_b = _required_env_path("IZ_T673_MOD_B", python_source=True)
    output_path = _required_env_path("IZ_T673_COST_OUT", python_source=False)
    if path_a == path_b:
        _fail("IZ_T673_MOD_A と IZ_T673_MOD_B は別ファイルでなければならない")
    if output_path in (path_a, path_b):
        _fail("IZ_T673_COST_OUT が入力 module を上書きしようとしている")
    if not output_path.parent.is_dir():
        _fail(f"IZ_T673_COST_OUT の親 directory が存在しない: {output_path.parent}")

    sha256_a = _sha256_file(path_a)
    sha256_b = _sha256_file(path_b)
    module_a = _load_module(path_a, "_iz_t673_cost_arm_a")
    module_b = _load_module(path_b, "_iz_t673_cost_arm_b")
    commit_a = _git_head(path_a)
    commit_b = _git_head(path_b)
    if commit_a != commit_b:
        _fail(f"A/B の commit OID が一致しない: A={commit_a} B={commit_b}")

    # 入力構築はすべて timed block の外。各 arm 固有の class だけを使う。
    cases_a = {m: _build_case(module_a, m) for m in M_VALUES}
    cases_b = {m: _build_case(module_b, m) for m in M_VALUES}
    order_rng = random.Random(20260810)
    bootstrap_rng = random.Random(20260811)
    measurement_started = time.monotonic()
    gc_initial = gc.isenabled()
    gc.disable()
    try:
        measurements = [
            _measure_m(
                m,
                cases_a[m],
                cases_b[m],
                order_rng,
                bootstrap_rng,
                measurement_started,
            )
            for m in M_VALUES
        ]
    finally:
        if gc_initial:
            gc.enable()
    measurement_finished = time.monotonic()
    if _sha256_file(path_a) != sha256_a or _sha256_file(path_b) != sha256_b:
        _fail("測定中に A/B module の内容が変化した")

    notes = [
        "outlier は一件も除外しない",
        "順位は三条件の連言で JSON 内だけに記録し pytest 合否には使わない",
        "実用上の許容閾値は設定せず、1 call/operation の換算だけを出す",
    ]
    for measurement in measurements:
        if measurement["budget_adjustment"] is not None:
            notes.append(str(measurement["budget_adjustment"]["reason"]))
    if all(measurement["budget_adjustment"] is None for measurement in measurements):
        notes.append("10 分超過予測はなく M=64 の K 低減は行わなかった")
    if measurement_finished - measurement_started > RUNTIME_LIMIT_SECONDS:
        notes.append("実測時間が 10 分上限を超えた")

    report = {
        "schema": "izanagi/t673-d-cost/v1",
        "hostname": socket.gethostname(),
        "python_version": platform.python_version(),
        "sys_executable": sys.executable,
        "commit_oid": commit_a,
        "modules": {
            "A": {
                "role": "base",
                "absolute_path": str(path_a),
                "sha256": sha256_a,
                "import_name": module_a.__name__,
            },
            "B": {
                "role": "D1",
                "absolute_path": str(path_b),
                "sha256": sha256_b,
                "import_name": module_b.__name__,
            },
        },
        "seeds": {
            "order": ORDER_SEED,
            "bootstrap": BOOTSTRAP_SEED,
        },
        "gc": {
            "initially_enabled": gc_initial,
            "enabled_during_measurement": False,
            "enabled_after_measurement": gc.isenabled(),
        },
        "measurement_started_monotonic": measurement_started,
        "measurement_finished_monotonic": measurement_finished,
        "measurement_elapsed_seconds": measurement_finished - measurement_started,
        "runtime_limit_seconds": RUNTIME_LIMIT_SECONDS,
        "measurements": measurements,
        "notes": notes,
    }
    output_path.write_text(
        json.dumps(report, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )
    try:
        written = json.loads(output_path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        _fail(f"測定 JSON を書込み後に検証できない: {exc}")
    if written.get("schema") != report["schema"]:
        _fail("測定 JSON の書込み後検証に失敗した")
