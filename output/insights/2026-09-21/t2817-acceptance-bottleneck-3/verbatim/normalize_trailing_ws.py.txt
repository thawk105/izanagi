#!/usr/bin/env python3
"""DW-S07 の可逆最小正規化: 指定 file の行末空白 (space/tab) を除き、除いた suffix を行番号つきで JSON に残す。
可視文字は不変。usage: normalize_trailing_ws.py <out.json> <file>..."""
import hashlib, json, sys
out = sys.argv[1]
record = {}
for path in sys.argv[2:]:
    raw = open(path, "rb").read()
    lines = raw.split(b"\n")
    changed = []
    new = []
    for i, line in enumerate(lines, 1):
        stripped = line.rstrip(b" \t")
        if stripped != line:
            changed.append([i, line[len(stripped):].decode("ascii")])
        new.append(stripped)
    new_raw = b"\n".join(new)
    if changed:
        open(path, "wb").write(new_raw)
    record[path.split("t2817-acceptance-bottleneck-3/", 1)[-1]] = {
        "original_bytes": len(raw),
        "original_sha256": hashlib.sha256(raw).hexdigest(),
        "normalized_bytes": len(new_raw),
        "normalized_sha256": hashlib.sha256(new_raw).hexdigest(),
        "stripped_lines": changed,
    }
json.dump(record, open(out, "w"), ensure_ascii=True, indent=1, sort_keys=True)
open(out, "a").write("\n")
for k, v in record.items():
    print(k, v["original_bytes"], "->", v["normalized_bytes"], "lines", len(v["stripped_lines"]))
