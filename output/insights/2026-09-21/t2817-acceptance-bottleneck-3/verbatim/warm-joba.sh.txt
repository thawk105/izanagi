#!/bin/bash
# [T-2817] Job A 用 worktree (同 SHA、別 dir) の pyc を login で温める (T-2710 job 6 の形、run_tests.py --collect-only)。
D=/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2817-acceptance-bottleneck-3
W=/work/1/SFC/tanab/izanagi/.claude/worktrees/dev-wave-t2817-joba
echo $$ > "$D/warm-joba.pid"
cd "$W" || { echo "rc=3" > "$D/warm-joba.done"; exit 3; }
date '+%Y-%m-%dT%H:%M:%S%z start' >> "$D/verbatim/pyc-warm-joba.times"
python3 tools/run_tests.py orchestrator/tests --collect-only -q -p no:cacheprovider > "$D/verbatim/pyc-warm-joba.log" 2>&1
rc=$?
date "+%Y-%m-%dT%H:%M:%S%z end rc=$rc" >> "$D/verbatim/pyc-warm-joba.times"
printf 'pyc=%s\npytest_pyc=%s\n' "$(find orchestrator tools -name '*.pyc' | wc -l)" "$(find orchestrator/tests -name '*pytest*.pyc' | wc -l)" >> "$D/verbatim/pyc-warm-joba.times"
echo "rc=$rc" > "$D/warm-joba.done"
exit "$rc"
