#!/bin/bash
# [T-2817] Job A (段階載せ collection 診断) の投入 launcher (親の運転 script、repo 外)。
# 計算ノード 1 job (generic) で wave 専用 dir に複製した probe を走らせる。cwd は dispatcher が投入元 worktree に固定する。
# 投入元は同 SHA の別 worktree dev-wave-t2817-joba (同一 worktree からの dispatch は直列で、Job B の orphan hold に当たるため)。
D=/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2817-acceptance-bottleneck-3
W=/work/1/SFC/tanab/izanagi/.claude/worktrees/dev-wave-t2817-joba
echo $$ > "$D/run-probe-a.pid"
export PYTHONDONTWRITEBYTECODE=1
cd "$W" || { echo "rc=3" > "$D/run-probe-a.done"; exit 3; }
date '+%Y-%m-%dT%H:%M:%S%z start' >> "$D/run-probe-a.times"
git rev-parse HEAD > "$D/run-probe-a.tip.txt"
python3 tools/pegasus/dispatch_compute.py --task generic \
  --walltime 00:40:00 --queue-wait-timeout 5400 --overall-grace 5400 \
  -- bash "$D/probe/t2817_collection_stage_probe.sh" > "$D/run-probe-a.log" 2>&1
rc=$?
date "+%Y-%m-%dT%H:%M:%S%z end rc=$rc" >> "$D/run-probe-a.times"
echo "rc=$rc" > "$D/run-probe-a.done"
exit "$rc"
