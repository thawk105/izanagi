#!/usr/bin/env python3
"""Job B の spans-<worker>-*.jsonl を時系列に並べる (read-only 整形)。usage: read_spans_worker.py <A dir> <worker>"""
import glob, json, sys
d, w = sys.argv[1], sys.argv[2]
names = ("get", "key_wait", "build", "copy", "runtime", "git", "history", "issue", "verify", "base_copy", "helper")
rows = []
for p in glob.glob(f"{d}/spans-{w}-*.jsonl"):
    for line in open(p):
        e = json.loads(line)
        if e["name"] in names:
            rows.append((e["begin_epoch"], e["name"], round(e["end_epoch"] - e["begin_epoch"], 3), e.get("key"), (e.get("extra") or {}).get("nodeid", "")[-75:]))
rows.sort()
for r in rows:
    print(r)
