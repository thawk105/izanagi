def finite(value):
    return (type(value) in (float, int) and value >= 0 and math.isfinite(value))


def load_ledger(path):
    data = json.loads(path.read_text(encoding='utf-8'))
    durations = data.get('duration_seconds_by_nodeid')
    if (type(data.get('schema_version')) is not int or data['schema_version'] != 1 or
            data.get('unit') != 'seconds' or not isinstance(durations, dict) or
            type(data.get('nodeid_count')) is not int or data['nodeid_count'] != len(durations)):
        raise ValueError('invalid duration ledger schema')
    return {k: float(v) for k, v in durations.items() if isinstance(k, str) and finite(v)}


def ledger_value(node, group, ledger):
    if node in ledger:
        return ledger[node]
    return ledger.get(node + '@' + group) if group else None


def runtime_nodeid(node, group):
    runtime = node + '@' + group if group else node
    # _real_repo_node_id: module basename + original function, no parameters.
    short = node.split('::')[0].rsplit('/', 1)[-1] + '::' + node.split('::')[1].split('[', 1)[0].split('@', 1)[0]
    if group == 'real-repo' and short in REAL_REPO_STRIP_NODES:
        runtime = node
    return runtime


def scope(node):
    return node.rsplit('@', 1)[-1] if node.rfind('@') > node.rfind(']') else node


def junit_medians(paths, selected, groups):
    # Invert pytest's module/class/name mangling against the selected universe.
    # No pairing property is read; parameters remain opaque strings.
    lookup = defaultdict(set)
    for node in selected:
        parts = node.split('::')
        module = parts[0].removesuffix('.py').replace('/', '.')
        classname = '.'.join([module] + parts[1:-1])
        for name in {parts[-1], parts[-1] + '@' + groups[node] if groups[node] else parts[-1]}:
            lookup[(classname, name)].add(node)
    samples = defaultdict(list)
    unmatched = []
    for path in paths:
        seen = set()
        for case in ET.parse(path).getroot().iter('testcase'):
            key = (case.get('classname', ''), case.get('name', ''))
            candidates = lookup.get(key, set())
            if not candidates:
                # A narrower pytest rootdir may omit a leading path component.
                candidates = set().union(*(nodes for (cls, name), nodes in lookup.items()
                                           if name == key[1] and cls.endswith('.' + key[0])))
            if len(candidates) != 1:
                unmatched.append({'file': str(path), 'classname': key[0], 'name': key[1],
                                  'reason': 'unmapped/ambiguous'})
                continue
            node = next(iter(candidates))
            if node in seen:
                raise ValueError(f'duplicate testcase for {node} in {path}')
            seen.add(node)
            value = float(case.attrib['time'])
            if not finite(value):
                raise ValueError(f'invalid junit time for {node}')
            samples[node].append(value)
    return {n: statistics.median(v) for n, v in samples.items()}, dict(samples), unmatched


def make_units(selected, groups, ledger):
    units = {}
    for node in selected:
        key = scope(runtime_nodeid(node, groups[node]))
        unit = units.setdefault(key, {'scope': key, 'index': len(units), 'nodes': []})
        unit['nodes'].append(node)
    for unit in units.values():
        costs = [ledger_value(n, groups[n], ledger) for n in unit['nodes']]
        unit['known'] = all(c is not None for c in costs) and math.isfinite(sum(c for c in costs if c is not None))
        unit['cost'] = sum(costs) if unit['known'] else None
    return list(units.values())


def order_units(units):
    known = sorted((u['cost'] for u in units if u['known']), reverse=True)
    if not known:
        raise ValueError('no known unit: policy default cost undefined; model unavailable')
    default = known[min(INITIAL_DISTRIBUTION_UNITS, len(known)) - 1]
    for u in units:
        if not u['known']:
            u['cost'] = default
    ordered = sorted(units, key=lambda u: (-u['cost'], u['index']))
    realized = sorted(ordered, key=lambda u: -len(u['nodes']))
    if len(realized) >= INITIAL_DISTRIBUTION_UNITS:
        candidates = list(enumerate(realized[WORKERS:], start=WORKERS))
        partners = sorted(candidates, key=lambda entry: (entry[1]['cost'], entry[0]))[:WORKERS]
        positions = {position for position, _ in partners}
        paired = (realized[:WORKERS] + [u for _, u in partners] +
                  [u for position, u in candidates if position not in positions])
        if [u['scope'] for u in sorted(paired, key=lambda u: -len(u['nodes']))] != [u['scope'] for u in paired]:
            raise ValueError('acceptance pairing infeasible: cardinality reorder would change intended queue')
        realized = paired
    # With fewer than 96 units conftest does not pair, but xdist still sorts cardinality.
    for rank, unit in enumerate(realized):
        unit['rank'] = rank
        unit['partner'] = len(realized) >= INITIAL_DISTRIBUTION_UNITS and WORKERS <= rank < 2 * WORKERS
    return realized, default


def simulate(ordered, durations):
    workers = [{'worker': i, 'occupancy': 0.0, 'units': []} for i in range(WORKERS)]
    heap = []
    assignments = {}
    for i, unit in enumerate(ordered):
        if i < WORKERS:
            start, worker = 0.0, i
        else:
            start, worker = heapq.heappop(heap)
        duration = sum(durations[n] for n in unit['nodes'])
        entry = {**unit, 'duration': duration, 'start': start, 'end': start + duration}
        workers[worker]['units'].append(entry)
        workers[worker]['occupancy'] = start + duration
        assignments[unit['scope']] = worker
        heapq.heappush(heap, (start + duration, worker))
    maximum = max(workers, key=lambda w: w['occupancy'])
    longest = max(ordered, key=lambda u: sum(durations[n] for n in u['nodes']))
    length = sum(durations[n] for n in longest['nodes'])
    longest_worker = workers[assignments[longest['scope']]]
    return {'O_max_model': maximum['occupancy'], 'maximum_worker': maximum,
            'L': length, 'longest_unit_scope': longest['scope'], 'L_worker': longest_worker,
            'O_max_model_minus_L': maximum['occupancy'] - length, 'workers': workers}


def model(ledger, report, junits):
    selected = report['selected']
    if (not isinstance(selected, list) or not selected or len(set(selected)) != len(selected) or
            not all(isinstance(n, str) and '::' in n for n in selected)):
        raise ValueError('selected must be a nonempty unique nodeid list')
    raw = report['observed_universe']
    groups = ({r['nodeid']: r['group'] for r in raw} if isinstance(raw, list) else dict(raw))
    if any(n not in groups or (groups[n] is not None and not isinstance(groups[n], str)) for n in selected):
        raise ValueError('selected node group missing/invalid')
    medians, samples, unmatched = junit_medians(junits, selected, groups)
    a_order, default = order_units(make_units(selected, groups, ledger))
    missing = [n for n in selected if ledger_value(n, groups[n], ledger) is None]
    refreshed = dict(ledger)
    refreshed.update({n: medians[n] for n in missing if n in medians})
    b_order, b_default = order_units(make_units(selected, groups, refreshed))
    durations = {}
    sources = {}
    for n in selected:
        prior = ledger_value(n, groups[n], ledger)
        durations[n] = medians[n] if n in medians else prior if prior is not None else default
        sources[n] = 'junit median' if n in medians else 'ledger' if prior is not None else 'baseline default'
    a, b = simulate(a_order, durations), simulate(b_order, durations)
    limitations = [
        '元 collection index は report に無く未再現。selected の保存順を unit index とし、同値はその index 順。',
        'real-repo suffix 除去は author 時 conftest registry の literal snapshot。将来の registry 変更は追随しない。',
        '動的 xdist の先読み・共有 base 相互作用・setup 重複は再現しない。指定の空き worker への list scheduling。',
        '同時に空いた worker は番号順。L が同値なら queue 上の最初を表示。',
        '所要は両案で固定。JUnit と ledger の両方に無い node は (a) の既定 cost を共用。',
    ]
    without_median = [n for n in missing if n not in medians]
    if without_median:
        limitations.append('(b) 未収載のうち JUnit 中央値なしは未補完: ' + str(len(without_median)) + ' node')
    return {'disclaimer': DISCLAIMER, 'workers': WORKERS, 'initial_distribution_units': INITIAL_DISTRIBUTION_UNITS,
            'limitations': limitations, 'a_current_ledger': {'default_cost': default, **a},
            'b_missing_with_junit_medians': {'default_cost': b_default, **b},
            'O_max_model_a_minus_b': a['O_max_model'] - b['O_max_model'],
            'missing_ledger_nodes': missing, 'missing_junit_for_missing_ledger': without_median,
            'junit_unmatched': unmatched, 'junit_medians': medians, 'junit_samples': samples,
            'duration_sources': sources, 'fixed_durations': durations}


def markdown(data):
    out = [DISCLAIMER, '\n| 案 | O_max_model | L | O_max_model − L | L worker 占有 |',
           '| --- | --- | --- | --- | --- |']
    for key in ('a_current_ledger', 'b_missing_with_junit_medians'):
        scenario = data[key]
        out.append(f"| {key} | {scenario['O_max_model']:.6f} | {scenario['L']:.6f} | {scenario['O_max_model_minus_L']:.6f} | {scenario['L_worker']['occupancy']:.6f} |")
    out.append(f"\n(a) − (b) = {data['O_max_model_a_minus_b']:.6f} 秒 (model)。rank は 0 起点。")
    for key in ('a_current_ledger', 'b_missing_with_junit_medians'):
        out.append('\n## ' + key)
        out.append(f"既定 cost = {data[key]['default_cost']:.6f}")
        for label in ('maximum_worker', 'L_worker'):
            w = data[key][label]
            out.extend([f"\n{label}: worker {w['worker']}, occupancy={w['occupancy']:.6f}",
                        '\n| rank | scope | cost | 所要 | start | end | nodes |',
                        '| --- | --- | --- | --- | --- | --- | --- |'])
            for u in w['units']:
                values = [u[k] for k in ('rank', 'scope', 'cost', 'duration', 'start', 'end', 'nodes')]
                out.append('| ' + ' | '.join(str(v).replace('|', '\\|') for v in values) + ' |')
    out.append('\n## 限界\n\n' + '\n'.join('- ' + v for v in data['limitations']))
    out.append('\nJUnit 未対応 testcase: ' + str(len(data['junit_unmatched'])))
    return '\n'.join(out) + '\n'


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--ledger', required=True, type=Path)
    parser.add_argument('--report', required=True, type=Path)
    parser.add_argument('--junit', required=True, action='append', type=Path)
    parser.add_argument('--markdown', required=True, type=Path)
    parser.add_argument('--json', required=True, type=Path)
    args = parser.parse_args()
    data = model(load_ledger(args.ledger), json.loads(args.report.read_text(encoding='utf-8')), args.junit)
    data['inputs'] = {'ledger': str(args.ledger), 'report': str(args.report), 'junit': list(map(str, args.junit))}
    args.json.write_text(json.dumps(data, ensure_ascii=False, indent=2, allow_nan=False) + '\n', encoding='utf-8')
    args.markdown.write_text(markdown(data), encoding='utf-8')


if __name__ == '__main__':
    main()
