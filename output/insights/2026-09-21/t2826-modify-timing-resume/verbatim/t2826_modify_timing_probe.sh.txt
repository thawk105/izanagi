#!/usr/bin/env bash
# T-2826 Job A: invoke only inside the allocated compute job.
set -u
WAVE_ROOT="$(pwd -P)"
PROBE_DIR="$(cd "$(dirname "$0")" && pwd -P)" || exit 2
export PYTHONDONTWRITEBYTECODE=1 IZANAGI_TASK_RUN_AUTO_RECORD=0 PYTHONPATH="$PROBE_DIR"
unset PYTEST_ADDOPTS PYTEST_XDIST_TESTRUNUID IZANAGI_TASK_RUN_SIDECAR PYTHONPYCACHEPREFIX
unset IZANAGI_TEST_RUNNER_EXCLUSIONS_V1 IZANAGI_ACCEPTANCE_SHARDS IZANAGI_ACCEPTANCE_SHARD_PLUGIN_V1
unset PYTEST_PLUGINS PYTEST_DISABLE_PLUGIN_AUTOLOAD IZANAGI_RUN_GROWTH_HELD_TESTS
unset T2817_PROBE_COLLECT_ONLY T2817_PROBE_OUT
unset T2826_PROBE_COLLECT_ONLY T2826_PROBE_OUT
export LC_ALL=C PYTHONUTF8=1
OUT_ROOT=${T2826_OUT:-/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2826-resume/job-out}
CELLS=${T2826_CELLS:-warm S1-a S2f-a S3u-a S3f-a S3cf-a S3cf-b S3f-b S3u-b S2f-b S1-b}
WORKERS=${T2826_WORKERS:-48}
for variable in ${!T2826_@}; do unset "$variable"; done
[[ $WORKERS =~ ^[1-9][0-9]*$ ]] || exit 2
PROBE_USER=$(id -un)
if mkdir -p "/scr/$PROBE_USER" 2>/dev/null && [[ -w /scr/$PROBE_USER ]]; then
    LOCAL=/scr/$PROBE_USER/t2826-$$
else
    LOCAL=/tmp/t2826-$$
fi
FINAL_RC=0
mkdir -p "$OUT_ROOT" "$(dirname "$LOCAL")" || exit 2
OUT_ROOT=$(cd "$OUT_ROOT" && pwd -P) || exit 2
# Do not merge a fresh run with old cells / a terminal marker.
if [[ -n $(ls -A "$OUT_ROOT") ]]; then
    echo 'OUT_ROOT must be empty' >&2
    exit 2
fi
mkdir "$LOCAL" || exit 2
stop_tree() {
    local parent=$1 child
    while read -r child; do
        [[ -n $child ]] || continue
        stop_tree "$child"
        kill -TERM "$child" 2>/dev/null || :
    done < <(awk -v p="$parent" '$2 == p {print $1}' <<< "$PROCESS_TREE")
}
pycache_counts() {
    printf 'pyc='
    find "$WAVE_ROOT/orchestrator" "$WAVE_ROOT/tools" -name '*.pyc' | wc -l
    printf 'pytest_pyc='
    find "$WAVE_ROOT/orchestrator" "$WAVE_ROOT/tools" -name '*pytest*.pyc' | wc -l
}
finish() {
    local rc=$? PROCESS_TREE
    trap - EXIT INT TERM HUP
    if (( rc != 0 )); then
        FINAL_RC=$rc
        PROCESS_TREE=$(ps -eo pid=,ppid=)
        stop_tree "$$"
        wait 2>/dev/null || :
    fi
    if [[ -d $LOCAL/out ]]; then
        pycache_counts > "$LOCAL/out/pycache-after.txt"
        rsync -a --exclude tmp "$LOCAL/out/" "$OUT_ROOT/" || FINAL_RC=2
    fi
    rm -rf -- "$LOCAL" || FINAL_RC=2
    printf 'rc=%s\n' "$FINAL_RC" > "$OUT_ROOT/probe.done" || FINAL_RC=2
    exit "$FINAL_RC"
}
trap finish EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
trap 'exit 129' HUP
mkdir -p "$LOCAL/out" "$LOCAL/tmp" || exit 2
export TMPDIR=$LOCAL/tmp
O=$LOCAL/out
elapsed() { awk -v a="$1" -v b="$2" 'BEGIN {printf "%.9f\n", b-a}'; }
lustre() {
    local f seen=0 failed=0
    for f in /proc/fs/lustre/llite/*/stats /proc/fs/lustre/mdc/*/md_stats; do
        [[ -e $f ]] || continue
        seen=1
        printf '### %s\n' "$f"
        cat "$f" || failed=1
    done
    if (( !seen || failed )); then echo 'lustre_stats=unreadable'; return 1; fi
}
singleton() {
    local dir=$1 short_user
    ps -eo user,pid,comm --no-headers > "$dir/singleton-before.txt"
    short_user=$(ps -o user= -p $$ | xargs)
    awk -v me="$PROBE_USER" -v short="$short_user" '
      $1 != me && $1 != short && $1 !~ /^(root|daemon|nqs.*|message.*|systemd.*|syslog|_.*|bin|sys|sync|games|man|lp|mail|news|uucp|proxy|www-data|backup|list|irc|gnats|nobody|sshd|ntp|chrony|rpc|rpcuser|dbus|polkitd|postfix|munge)$/ {n++}
      END {print "others=" n+0}' "$dir/singleton-before.txt"
    ps -eo uid=,pid=,comm=,args= > "$dir/process-args-before.txt"
    awk -v uid="$(id -u)" '$1 == uid && ($3 ~ /^pytest/ || ($3 ~ /^python/ && $0 ~ / -m pytest( |$)/)) {n++}
      END {print "stale_pytest=" n+0}' "$dir/process-args-before.txt"
}
snapshot() {
    local dir=$1 phase=$2
    cat /proc/loadavg > "$dir/loadavg-$phase.txt"
    awk '/^MemAvailable:/' /proc/meminfo > "$dir/mem-$phase.txt"
    lustre > "$dir/lustre-$phase.txt" 2>&1 || echo 'lustre_stats=unreadable' >> "$dir/cell.txt"
}
LUSTRE=readable
lustre > "$O/lustre-initial.txt" 2>&1 || LUSTRE=unreadable
{
    hostname
    date --iso-8601=seconds
    uname -r
    nproc
    free -g
    df -h /scr /tmp "$WAVE_ROOT"
    mount | grep -E ' /scr | /tmp | /work '
    python3.10 --version
    python3.10 -m pytest --version
    python3.10 -c 'import xdist; print(xdist.__version__)'
    printf 'lustre_stats=%s\nWAVE_ROOT=%s\n' "$LUSTRE" "$WAVE_ROOT"
    git -C "$WAVE_ROOT" rev-parse HEAD
    git -C "$WAVE_ROOT" status --porcelain | wc -l
    pycache_counts
    taskset -pc $$
    cat /proc/loadavg
} > "$O/env.txt" 2>&1
python3.10 -c 'import t2826_probe_plugin' > "$O/plugin-import.out" 2> "$O/plugin-import.err" || exit 3

run_cell() {
    local name=$1 stage=${1%-*} half=${1##*-} start end rc=0 escaped variable
    local CELL=$O/$name
    local -a command=(python3.10 -m pytest "$WAVE_ROOT/orchestrator/tests")
    [[ $name == warm ]] && half=warm
    mkdir "$CELL" || exit 2
    mkdir "$CELL/tmp" || exit 2
    export TMPDIR=$CELL/tmp
    unset IZANAGI_ACCEPTANCE_SHARD_PLUGIN_V1
    for variable in ${!T2826_@}; do unset "$variable"; done
    if [[ $stage == warm ]]; then
        command+=(--collect-only -q -p no:cacheprovider)
    else
        export T2826_PROBE_COLLECT_ONLY=1 T2826_PROBE_OUT=$CELL/probe.json
        command+=(-n "$WORKERS" --dist loadgroup)
        [[ $stage == S2f || $stage == S3f || $stage == S3cf ]] && export T2826_FUNC_TIMERS=1
        [[ $stage == S3cf ]] && export T2826_RESOLVE_MEMO=1
        [[ $stage == S1 || $stage == S2f ]] && command+=(--no-loadscope-reorder)
        command+=("--junitxml=$CELL/junit.xml")
        if [[ $stage == S2f || $stage == S3* ]]; then
            mkdir -p "$CELL/session/shard-0" || exit 2
            # JSON escaping is done by bash; exactly the three spec keys.
            escaped=${CELL//\\/\\\\}
            escaped=${escaped//\"/\\\"}
            printf -v IZANAGI_ACCEPTANCE_SHARD_PLUGIN_V1 '{"session_root":"%s/session","shard_count":3,"shard_index":0}' "$escaped"
            export IZANAGI_ACCEPTANCE_SHARD_PLUGIN_V1
            command+=(-p tools.acceptance_shards -p no:cacheprovider -p t2826_probe_plugin)
        else
            command+=(-p no:cacheprovider -p t2826_probe_plugin)
        fi
    fi
    start=$(date +%s.%N)
    {
        printf 'cell=%s\nstage=%s\nhalf=%s\nstart_epoch=%s\ncwd=%s\nargv=' "$name" "$stage" "$half" "$start" "$WAVE_ROOT"
        printf '%q ' "${command[@]}"
        printf '\ntmpdir=%s\nworkers=%s\n' "$TMPDIR" "$WORKERS"
        singleton "$CELL"
    } > "$CELL/cell.txt"
    snapshot "$CELL" before
    /usr/bin/time -f '%e %U %S %M' -o "$CELL/time.txt" "${command[@]}" > "$CELL/out" 2> "$CELL/err"
    rc=$?
    end=$(date +%s.%N)
    snapshot "$CELL" after
    ps -eo user,pid,comm --no-headers > "$CELL/singleton-after.txt"
    ps -eo uid=,pid=,comm=,args= > "$CELL/process-args-after.txt"
    printf 'memo_files_begin\n' >> "$CELL/cell.txt"
    find "$TMPDIR" -maxdepth 4 -type f \( -name '*.pending' -o -name '*.failed' -o -iname '*memo*' \) -printf '%P\n' >> "$CELL/cell.txt"
    printf 'memo_files_end\n' >> "$CELL/cell.txt"
    (( rc == 0 )) || FINAL_RC=$rc
    printf 'end_epoch=%s\nwall=%s\nrc=%s\n' "$end" "$(elapsed "$start" "$end")" "$rc" >> "$CELL/cell.txt"
    unset IZANAGI_ACCEPTANCE_SHARD_PLUGIN_V1
    for variable in ${!T2826_@}; do unset "$variable"; done
    mkdir -p "$OUT_ROOT/$name" || exit 2
    rsync -a --exclude tmp "$CELL/" "$OUT_ROOT/$name/" || FINAL_RC=2
    [[ $stage == warm && $rc != 0 ]] && exit 4
    return 0
}
for cell in $CELLS; do
    case "$cell" in warm|S1-[ab]|S2f-[ab]|S3u-[ab]|S3f-[ab]|S3cf-[ab]) ;; *) exit 2 ;; esac
    run_cell "$cell"
done
exit "$FINAL_RC"
