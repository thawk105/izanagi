#!/usr/bin/env python3
"""[T-2632] commit 群の変異を 1 件だけ作業木へ注入する (親専用、repo に入れない)。

usage: apply_mutant.py <spec.json> <mutation-id> <repo-root>
spec の replacements を累積適用し、各置換の old が累積後の内容で exactly 1 回だけ現れることを assert する (DW-M04)。
注入後の変更 file 集合が replacements の file 集合と一致することは呼び手 (run-commit-group.sh) が git で検査する。
"""
import json
import pathlib
import sys


def main(argv):
    spec = json.loads(pathlib.Path(argv[1]).read_text(encoding="utf-8"))
    mid, root = argv[2], pathlib.Path(argv[3])
    matches = [m for m in spec["mutations"] if m["id"] == mid]
    if len(matches) != 1:
        raise SystemExit(f"mutation id not unique in spec: {mid} ({len(matches)})")
    contents = {}
    for rep in matches[0]["replacements"]:
        path = root / rep["file"]
        text = contents.get(rep["file"])
        if text is None:
            text = path.read_text(encoding="utf-8")
        count = text.count(rep["old"])
        if count != 1:
            raise SystemExit(f"{mid}: old occurs {count} times in {rep['file']}")
        text = text.replace(rep["old"], rep["new"], 1)
        contents[rep["file"]] = text
    for rel, text in contents.items():
        (root / rel).write_text(text, encoding="utf-8")
    print(json.dumps({"id": mid, "files": sorted(contents)}))


if __name__ == "__main__":
    main(sys.argv)
