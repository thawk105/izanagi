MUTATIONS["e1"] = [m for m in MUTATIONS["loop"] if m[0].startswith("E1-")]
# 段 6 裁定 §5 の割り振り: 注入群 L (drift を受けない新設 test で検出) と commit 群 (drift を受ける test でだけ検出)。
_LINJ = ("S3-", "S4-", "S5-", "S6-", "S7-", "S8-", "S9-", "E1-")
_CG = ("S1-", "S2-", "S10-", "S11-", "S12-", "S13-", "S14-", "S15-")
MUTATIONS["linj"] = [m for m in MUTATIONS["loop"] if m[0].startswith(_LINJ)]
MUTATIONS["cg"] = [m for m in MUTATIONS["loop"] if m[0].startswith(_CG)]


def build(group, expected):
    out = []
    for mid, category, reps in MUTATIONS[group]:
        nodes = [] if expected is None else sorted(expected.get(mid, []))
        out.append({
            "id": mid,
            "category": category,
            "replacements": [{"file": f, "old": o, "new": n} for f, o, n in reps],
            "expected_nodes": nodes,
            "expected_status": "SURVIVED" if not nodes else "KILLED",
            "hang_risk": False,
        })
    return {
        "schema": "izanagi-dev-wave-mutation-spec/v1",
        "estimated_run_seconds": 240,
        "hang_timeout_seconds": 3000,
        "timeout_seconds": 9000,
        "mutations": out,
    }


def main(argv):
    group, out_path = argv[1], pathlib.Path(argv[2])
    expected = json.loads(pathlib.Path(argv[3]).read_text(encoding="utf-8")) if len(argv) > 3 else None
    spec = build(group, expected)
    raw = (json.dumps(spec, ensure_ascii=False, indent=1, sort_keys=True) + "\n").encode("utf-8")
    out_path.write_bytes(raw)
    print(hashlib.sha256(raw).hexdigest())


if __name__ == "__main__":
    main(sys.argv)
