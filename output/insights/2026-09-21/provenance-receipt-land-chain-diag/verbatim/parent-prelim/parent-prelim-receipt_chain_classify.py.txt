#!/usr/bin/env python3
"""受領証列 (共有 store) から、各受領証を発行した走が warm (祖先受領証を再利用できた) か
cold (できなかった) かを推定し、cold の原因を bindings の差分項目で分類する。

read-only。判定・schema には触れない。祖先判定は git merge-base --is-ancestor。

推定の根拠 (tools/check_ai_provenance.py の _audit_history / _receipt_prefix):
  - lookup は同 partition (= environment digest の directory) の受領証のうち tip が HEAD の祖先の
    ものを距離順に並べ、bindings が完全一致し selection/delta/coverage 等が整合する最初の 1 件を再利用する。
  - したがって「同 partition に、mtime がこの走より前で、tip が祖先 (または同一) で、bindings が完全一致する
    受領証がある」なら warm と推定できる (selection/delta の整合は ff-only 履歴で通常成立)。
  - 無ければ cold。原因 = (a) 同 partition に祖先受領証が無い → 他 partition にあれば「partition 跨ぎ」、
    無ければ「初回 / 祖先受領証なし」; (b) 祖先受領証はあるが bindings が違う → 差分項目で分類。
  - prune (64 件上限) で消えた受領証は見えない。partition の件数が 64 なら「祖先なし」は prune の可能性を含む。
"""
import hashlib
import json
import subprocess
import sys
from collections import Counter
from datetime import datetime, timezone, timedelta
from pathlib import Path

STORE = Path(sys.argv[1])
REPO = Path(sys.argv[2])
SINCE = sys.argv[3] if len(sys.argv) > 3 else "2026-09-20T21:00:00"
OUT = Path(sys.argv[4]) if len(sys.argv) > 4 else None
JST = timezone(timedelta(hours=9))
since_ns = int(datetime.fromisoformat(SINCE).replace(tzinfo=JST).timestamp() * 1e9)


def digest(value):
    return hashlib.sha256(json.dumps(
        value, sort_keys=True, ensure_ascii=True, separators=(",", ":"),
    ).encode("ascii")).hexdigest()


_anc_cache = {}


def is_ancestor(a, b):
    """a が b の祖先 (または同一) か。"""
    if a == b:
        return True
    key = (a, b)
    if key not in _anc_cache:
        r = subprocess.run(["git", "merge-base", "--is-ancestor", a, b], cwd=REPO,
                           stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        _anc_cache[key] = (r.returncode == 0)
    return _anc_cache[key]


def commit_count(a, b):
    r = subprocess.run(["git", "rev-list", "--count", f"{a}..{b}"], cwd=REPO,
                       stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True)
    return int(r.stdout.strip() or -1) if r.returncode == 0 else -1


def commit_time(oid):
    r = subprocess.run(["git", "show", "-s", "--format=%ci", oid], cwd=REPO,
                       stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True)
    return r.stdout.strip()[:16] if r.returncode == 0 else "?"


# 全受領証を読む (partition 判定には全件が要る)
all_rows = []
part_sizes = Counter()
for part in sorted(STORE.iterdir()):
    if not part.is_dir():
        continue
    for f in part.iterdir():
        if not f.is_file() or not f.name.endswith(".json"):
            continue
        st = f.lstat()
        try:
            r = json.loads(f.read_text(encoding="utf-8"))
        except Exception:  # noqa: BLE001
            continue
        part_sizes[part.name] += 1
        all_rows.append({
            "partition": part.name,
            "mtime_ns": st.st_mtime_ns,
            "mtime": datetime.fromtimestamp(st.st_mtime, JST).strftime("%m-%d %H:%M:%S"),
            "tip": r["tip"],
            "bindings": r["bindings"],
            "sel_count": r.get("selection", {}).get("count"),
        })
all_rows.sort(key=lambda x: x["mtime_ns"])

BIND_KEYS = ["attributes", "registry_manifest", "cab_hits", "policy",
             "scope_epoch", "implementation_epoch", "repository", "object_format",
             "environment"]


def bind_diff(a, b):
    out = []
    for k in BIND_KEYS:
        if a.get(k) != b.get(k):
            out.append(k)
    return out


results = []
for i, row in enumerate(all_rows):
    if row["mtime_ns"] < since_ns:
        continue
    earlier = [x for x in all_rows[:i] if x["mtime_ns"] < row["mtime_ns"]]
    same_part = [x for x in earlier if x["partition"] == row["partition"]]
    anc_same = [x for x in same_part if is_ancestor(x["tip"], row["tip"])]
    verdict = None
    cause = None
    detail = {}
    if anc_same:
        exact = [x for x in anc_same if x["bindings"] == row["bindings"]]
        if exact:
            verdict = "warm(推定)"
            # 最近接 (commit 距離最小)
            best = min(exact, key=lambda x: commit_count(x["tip"], row["tip"]))
            detail = {"reused_tip": best["tip"][:12], "reused_mtime": best["mtime"],
                      "delta_commits": commit_count(best["tip"], row["tip"])}
        else:
            verdict = "cold"
            best = min(anc_same, key=lambda x: commit_count(x["tip"], row["tip"]))
            diff = bind_diff(best["bindings"], row["bindings"])
            cause = "bindings差: " + ",".join(diff)
            detail = {"nearest_tip": best["tip"][:12], "nearest_mtime": best["mtime"],
                      "delta_commits": commit_count(best["tip"], row["tip"]),
                      "diff": diff}
            if "cab_hits" in diff:
                a = set(best["bindings"].get("cab_hits", []))
                b = set(row["bindings"].get("cab_hits", []))
                detail["cab_added"] = sorted(x[:12] for x in b - a)
                detail["cab_removed"] = sorted(x[:12] for x in a - b)
    else:
        verdict = "cold"
        anc_other = [x for x in earlier if x["partition"] != row["partition"]
                     and is_ancestor(x["tip"], row["tip"])]
        if anc_other:
            # 他 partition に祖先受領証あり → partition 跨ぎ。その partition との環境差を出す
            best = max(anc_other, key=lambda x: x["mtime_ns"])
            env_a, env_b = best["bindings"]["environment"], row["bindings"]["environment"]
            env_diff = [k for k in ("checker", "schema", "config", "inherited")
                        if env_a.get(k) != env_b.get(k)]
            if "checker" in env_diff:
                cause = "checker sha 変更 (新 partition)"
            else:
                cause = "partition 跨ぎ: env差=" + ",".join(env_diff)
            detail = {"other_partition": best["partition"][:12], "other_tip": best["tip"][:12],
                      "other_mtime": best["mtime"], "env_diff": env_diff}
            if "inherited" in env_diff:
                ia, ib = env_a.get("inherited", {}), env_b.get("inherited", {})
                detail["inherited_diff"] = {k: (ia.get(k), ib.get(k)) for k in set(ia) | set(ib)
                                            if ia.get(k) != ib.get(k)}
            if "config" in env_diff:
                la = set(env_a.get("config", "").splitlines())
                lb = set(env_b.get("config", "").splitlines())
                detail["config_diff"] = {"only_other": sorted(la - lb)[:6], "only_this": sorted(lb - la)[:6]}
            if "checker" in env_diff:
                detail["checker"] = (env_a.get("checker", "")[:12], env_b.get("checker", "")[:12])
        else:
            cause = "祖先受領証なし (初回)" + (" [prune の可能性]" if part_sizes[row["partition"]] >= 64 else "")
    results.append({
        "mtime": row["mtime"], "partition": row["partition"][:12], "tip": row["tip"][:12],
        "tip_time": commit_time(row["tip"]), "sel_count": row["sel_count"],
        "checker": row["bindings"]["environment"]["checker"][:12],
        "inherited_LANG": row["bindings"]["environment"].get("inherited", {}).get("LANG"),
        "GIT_EDITOR": row["bindings"]["environment"].get("inherited", {}).get("GIT_EDITOR"),
        "verdict": verdict, "cause": cause, **detail,
    })

if OUT:
    OUT.write_text("\n".join(json.dumps(r, ensure_ascii=False) for r in results) + "\n")

print(f"since={SINCE} rows={len(results)}")
print("verdict/cause counts:")
for (v, c), n in sorted(Counter((r["verdict"], r["cause"]) for r in results).items()):
    print(f"  {n:3d}  {v:10s} {c}")
print()
print("timeline:")
for r in results:
    extra = ""
    if r["verdict"].startswith("warm"):
        extra = f"reuse={r['reused_tip']} Δ={r['delta_commits']}"
    elif "diff" in r:
        extra = f"nearest={r['nearest_tip']} Δ={r['delta_commits']} diff={r['diff']}"
        if r.get("cab_added") or r.get("cab_removed"):
            extra += f" cab+={r.get('cab_added')} cab-={r.get('cab_removed')}"
    elif "other_partition" in r:
        extra = f"other_part={r['other_partition']} env_diff={r['env_diff']}"
        if r.get("inherited_diff"):
            extra += f" inh={r['inherited_diff']}"
        if r.get("config_diff"):
            extra += f" cfg={r['config_diff']}"
        if r.get("checker"):
            extra += f" checker={r['checker']}"
    print(f"{r['mtime']} part={r['partition']} tip={r['tip']} ({r['tip_time']}) n={r['sel_count']} "
          f"chk={r['checker']} LANG={r['inherited_LANG']} ED={r['GIT_EDITOR']} "
          f"{r['verdict']:10s} {r['cause'] or ''} {extra}")
