#!/usr/bin/env python3
"""Read-only main checker history from retained reflog, using only stdlib.

Reflog order is reversed as recorded, not sorted by commit or wall-clock time.
Version presence means a byte hash observed in ANY retained main reflog entry;
absence cannot prove that main never held a version before expiry. Raw reflog
old/new links detect some gaps, but cannot prove no entries were expired.
Marker line counts describe literal bytes, not a semantic implementation audit.
"""
import argparse
import datetime as dt
import hashlib
import json
from pathlib import Path
import subprocess
import sys

JST = dt.timezone(dt.timedelta(hours=9))
CHECKER = 'tools/check_ai_provenance.py'
REF = 'refs/heads/main'


def instant(value):
    parsed = dt.datetime.fromisoformat(value.replace('Z', '+00:00'))
    if parsed.tzinfo is None:
        raise ValueError('ISO time must include a timezone')
    return parsed


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument('--repo', required=True, type=Path)
    ap.add_argument('--since', required=True, type=instant)
    ap.add_argument('--commits', nargs='+', default=[])
    ap.add_argument('--out', required=True, type=Path)
    args = ap.parse_args()
    repo = args.repo.resolve()

    def git(*argv, accepted=(0,)):
        proc = subprocess.run(['git', '-C', str(repo), *argv], capture_output=True)
        if proc.returncode not in accepted:
            raise RuntimeError(f'git {argv}: rc={proc.returncode}: '
                               + proc.stderr.decode(errors='replace').strip())
        return proc

    def text(*argv):
        return git(*argv).stdout.decode().strip()

    def git_path(*argv):
        path = Path(text('rev-parse', *argv))
        return (repo / path).resolve()

    common = git_path('--git-common-dir')
    reflog_path = git_path('--git-path', 'logs/' + REF)
    shared = reflog_path == common / 'logs' / REF
    if not shared:
        raise RuntimeError(f'Unexpected non-shared main reflog: {reflog_path}')
    tip_before = text('rev-parse', '--verify', REF)
    raw_before = reflog_path.read_bytes()
    lines = git('reflog', 'show', '--date=iso-strict',
                '--format=%H%x09%gd%x09%gs', REF).stdout.decode().splitlines()
    entries = []
    for ordinal, line in enumerate(reversed(lines)):
        commit, selector, subject = line.split('\t', 2)
        timestamp = instant(selector.rsplit('@{', 1)[1][:-1])
        entries.append(dict(type='entry', ordinal=ordinal, commit=commit,
                            time=timestamp.astimezone(JST).isoformat(),
                            selector=selector, subject=subject))
    if not entries:
        raise RuntimeError('No retained main reflog entries')
    resolved = [(c, text('rev-parse', '--verify', '--end-of-options', c+'^{commit}'))
                for c in args.commits]
    commits = list(dict.fromkeys([e['commit'] for e in entries]
                                + [c for _, c in resolved]))
    specs = ''.join(c + ':' + CHECKER + '\n' for c in commits).encode()
    # One batch lookup avoids reading identical checker bytes thousands of times.
    proc = subprocess.run(['git', '-C', str(repo), 'cat-file',
                           '--batch-check=%(objectname) %(objecttype)'],
                          input=specs, capture_output=True, check=True)
    blob_lines = proc.stdout.decode().splitlines()
    if len(blob_lines) != len(commits):
        raise RuntimeError('Incomplete cat-file batch result')
    cache, versions = {}, {}
    for commit, line in zip(commits, blob_lines):
        oid, kind = line.rsplit(' ', 1)
        if kind != 'blob':
            versions[commit] = dict(sha256=None, error=line)
            continue
        if oid not in cache:
            data = git('cat-file', 'blob', oid).stdout
            cache[oid] = dict(sha256=hashlib.sha256(data).hexdigest(),
                             d2192_lines=sum(b'diff-merges=first-parent' in x
                                             for x in data.splitlines()),
                             outer_deadline_lines=sum(b'OUTER_DEADLINE' in x
                                                      for x in data.splitlines()))
        versions[commit] = cache[oid]
    gaps = []
    raw_rows = raw_before.splitlines()
    last_new = None
    for n, row in enumerate(raw_rows):
        old, new = row.split(b' ', 2)[:2]
        if last_new is not None and old != last_new:
            def raw_time(raw):
                epoch = int(raw.split(b'\t', 1)[0].rsplit(b' ', 2)[1])
                return dt.datetime.fromtimestamp(epoch, JST).isoformat()
            gaps.append(dict(raw_index=n, previous_time=raw_time(raw_rows[n-1]),
                             next_time=raw_time(row), previous_new=last_new.decode(),
                             next_old=old.decode()))
        last_new = new
    monotonic = all(instant(a['time']) <= instant(b['time'])
                    for a, b in zip(entries, entries[1:]))
    first_in = next((i for i, e in enumerate(entries)
                     if instant(e['time']) >= args.since), len(entries))
    baseline = entries[first_in - 1] if first_in else None
    observed = {}
    transitions, unknown, selected = 0, 0, 0
    for i, entry in enumerate(entries):
        entry.update(versions[entry['commit']])
        digest = entry['sha256']
        if digest:
            observed.setdefault(digest, []).append(entry)
        entry['in_window'] = instant(entry['time']) >= args.since
        entry['baseline'] = entry is baseline
        previous = entries[i-1] if i else None
        old = previous['sha256'] if previous else None
        entry['old_sha256'] = old
        entry['transition'] = (digest != old) if digest and old else None
        if entry['in_window']:
            selected += 1
            transitions += entry['transition'] is True
            unknown += entry['transition'] is None
    queries = []
    for requested, commit in resolved:
        version = versions[commit]
        matches = observed.get(version['sha256'], [])
        rc = git('merge-base', '--is-ancestor', commit, tip_before,
                 accepted=(0, 1)).returncode
        queries.append(dict(type='commit', requested=requested, commit=commit,
                            **version, ancestor_of_main=rc == 0,
                            version_seen_on_main=bool(matches) if version['sha256'] else None,
                            matching_entries=len(matches),
                            first_match=matches[0]['time'] if matches else None))
    stable = (raw_before == reflog_path.read_bytes()
              and tip_before == text('rev-parse', '--verify', REF))
    coverage = dict(type='coverage', repo=str(repo), common_git_dir=str(common),
                    reflog_path=str(reflog_path), shared_main_reflog=shared,
                    main_tip=tip_before, since=args.since.astimezone(JST).isoformat(),
                    oldest=entries[0]['time'], newest=entries[-1]['time'],
                    raw_entries=len(raw_rows), parsed_entries=len(entries),
                    chain_gaps=gaps, timestamps_monotonic=monotonic,
                    stable_snapshot=stable, baseline_available=baseline is not None,
                    limitation='Retained reflog only; expiry/deletions before or within '
                    'this range cannot be fully ruled out. No match is not never-on-main.')
    summary = dict(type='summary', retained_entries=len(entries), window_entries=selected,
                   baseline_entries=int(baseline is not None), checker_transitions=transitions,
                   unknown_transitions=unknown, queried_commits=len(queries),
                   matched_versions=sum(q['version_seen_on_main'] is True for q in queries),
                   unreadable_checker_entries=sum(e['sha256'] is None for e in entries))
    args.out.parent.mkdir(parents=True, exist_ok=True)
    with args.out.open('w') as out:
        for row in [coverage, *entries, *queries, summary]:
            out.write(json.dumps(row, ensure_ascii=False) + '\n')

    def emit(*cells):
        print('\t'.join(str(x).replace('\t', '\\t').replace('\n', '\\n') for x in cells))

    def short(value):
        return value[:12] if value else 'unknown'

    emit('COVERAGE', coverage['oldest'], coverage['newest'])
    emit('MAIN_REFLOG', reflog_path, 'shared=' + str(shared).lower())
    emit('SNAPSHOT', 'stable=' + str(stable).lower(),
         'chain_gaps=' + str(len(gaps)), 'monotonic=' + str(monotonic).lower())
    emit('LIMIT', coverage['limitation'])
    for gap in gaps:
        emit('CHAIN_GAP', gap['previous_time'], gap['next_time'],
             gap['previous_new'], gap['next_old'])
    emit('kind', 'reflog_time_JST', 'new_commit', 'old_sha256', 'new_sha256', 'subject')
    if baseline is None:
        emit('baseline', 'unavailable (no retained entry before --since)')
    for entry in entries:
        if entry['baseline'] or (entry['in_window'] and entry['transition'] is not False):
            kind = 'baseline' if entry['baseline'] else ('transition' if entry['transition'] else 'unknown')
            emit(kind, entry['time'], entry['commit'],
                 '-' if entry['baseline'] else short(entry['old_sha256']),
                 short(entry['sha256']), entry['subject'])
    emit('commit', 'checker_sha256', 'd2192_lines', 'outer_deadline_lines',
         'ancestor_of_main', 'version_seen_on_retained_main', 'matching_entries', 'first_match_JST')
    for q in queries:
        emit(q['requested'], short(q['sha256']), q.get('d2192_lines', 'unknown'),
             q.get('outer_deadline_lines', 'unknown'), q['ancestor_of_main'],
             q['version_seen_on_main'], q['matching_entries'], q['first_match'] or '-')
    emit('SUMMARY', *(f'{k}={v}' for k, v in summary.items() if k != 'type'))
    return 0 if stable and monotonic and not gaps and not summary['unreadable_checker_entries'] else 1


if __name__ == '__main__':
    try:
        sys.exit(main())
    except (OSError, ValueError, RuntimeError, subprocess.SubprocessError) as exc:
        print(f'ERROR: {exc}', file=sys.stderr)
        sys.exit(2)
