#!/usr/bin/env python3
"""周期 (dispatcher poll 5 秒・producer 待ち手 poll 5 秒) を 0 にしたときの wave あたり短縮上限を、観測表から積算する。

直列経路の数え方:
  - 焦点走・単発 dispatch: 1 job = 1 producer → spool→receipt と done→waiter の両方
  - 受入: shard 3 本は並列。最後に END した shard の spool→receipt だけが直列経路 (+ done→waiter)
  - 変異: attempt は直列 (1 attempt ずつ dispatch) → 全 attempt の spool→receipt を積算 (+ 最終 attempt の done→waiter)
上限 = 観測された poll 由来区間の実測和 (poll を 0 にしても他の費用は残るので「上限」)。
"""
from __future__ import annotations

import json
import statistics as st
from collections import defaultdict
from pathlib import Path

HERE = Path(__file__).parent
rows = json.load(open(HERE / "collect_latency.rows.json"))


def f(x):
    return None if x in (None, "None", "") else float(x)


for r in rows:
    for k in ("started", "ended", "result", "spool", "receipt", "done", "waiter"):
        r[k] = f(r.get(k))

ROOT = Path("/work/1/SFC/tanab/dev-wave-jobs")


def waiter_is_receipt_json(r) -> bool:
    """collect_latency.py が選んだ待ち手 file が dev_wave_wait.py の receipt json か (mtime 一致で逆引き)。"""
    job = ROOT / r["wave"]
    producer = r["producer"]
    if "/" in producer:
        sub, _ = producer.split("/", 1)
        job = job / sub
    for c in list(job.glob("*.wait-receipt.json")) + list(job.glob("*-wait-receipt.json")) + list(job.glob("*-wait.json")):
        if abs(c.stat().st_mtime - r["waiter"]) < 1:
            return True
    return False


def is_serial_second_tag(r) -> bool:
    return r["wave"].endswith("t2807-b8-prerun") and r["producer"].endswith("g_rt")


per_wave = defaultdict(lambda: dict(n_serial=0, disp_poll=0.0, waiter_poll=0.0, n_waiter=0, n_focus=0, n_acc=0, n_mut=0))
# 受入は producer ごとに最後の shard だけ
acc_last: dict[tuple, dict] = {}
for r in rows:
    if r["kind"] == "acceptance" and r["ended"] is not None:
        key = (r["wave"], r["producer"])
        if key not in acc_last or r["ended"] > acc_last[key]["ended"]:
            acc_last[key] = r
serial_rows = [r for r in rows if r["kind"] in ("focus", "log", "mutation") and r["ended"] is not None] + list(acc_last.values())
for r in serial_rows:
    w = per_wave[r["wave"]]
    w["n_serial"] += 1
    w["n_" + ("focus" if r["kind"] in ("focus", "log") else "acc" if r["kind"] == "acceptance" else "mut")] += 1
    # dispatcher poll 由来: spool → receipt (spool が無い焦点走は END → receipt − 10 秒の代理)
    if r["spool"] is not None and r["receipt"] is not None:
        w["disp_poll"] += max(0.0, r["receipt"] - r["spool"])
    elif r["receipt"] is not None:
        w["disp_poll"] += max(0.0, r["receipt"] - r["ended"] - 10.0)
    # 待ち手 poll 由来: done → waiter (dev_wave_wait.py producer の receipt json で戻り時刻が取れた producer だけ。
    # 自前 wrapper (wait-*.done) や直列 wait-all の 2 本目は待ち手の poll ではないので除外する)
    if r["done"] is not None and r["waiter"] is not None and waiter_is_receipt_json(r) and not is_serial_second_tag(r):
        w["waiter_poll"] += max(0.0, r["waiter"] - r["done"])
        w["n_waiter"] += 1

print("wave                                  serial jobs (focus/acc/mut)  dispatcher poll 由来 [s]  待ち手 poll 由来 [s] (n)  合計 [s]")
tot = []
for wv, w in sorted(per_wave.items()):
    total = w["disp_poll"] + w["waiter_poll"]
    tot.append(total)
    print(f"{wv.replace('dev-wave-', ''):36s} {w['n_serial']:4d} ({w['n_focus']}/{w['n_acc']}/{w['n_mut']})   {w['disp_poll']:8.0f}   {w['waiter_poll']:8.0f} ({w['n_waiter']})   {total:8.0f}")
print(f"平均 {st.mean(tot):.0f} 秒 / 最大 {max(tot):.0f} 秒 (wave 所要 平均 154 分 = 9240 秒に対し {st.mean(tot)/9240*100:.1f}% / {max(tot)/9240*100:.1f}%)")
print("注: 待ち手 poll 由来は dev_wave_wait.py producer の receipt json で戻り時刻が取れた producer だけ積算 (mtime 一致で逆引き。自前 wrapper wait-*.done・直列 wait-all の 2 本目・receipt 不在は除外)。dispatcher poll 由来は spool が無い焦点走・単発で END→receipt 相当−10 秒を代理にした。")
