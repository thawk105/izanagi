#!/usr/bin/env python3
"""[T-2797] probe2 (M4 以外) と probe3 (M4-rr95、M0) の観測 node から observed-nodes.json を作る (DW-M08、dispatch probe の観測)。"""
import json
import pathlib

J = pathlib.Path("/work/1/SFC/tanab/dev-wave-jobs/dev-wave-t2797-tier0")


def nodes(results):
    d = json.loads((J / results).read_text())
    return {m["id"]: sorted(set(m.get("failed_nodes") or [])) for m in d["mutations"]}


p2, p3 = nodes("mutation-probe2-results.json"), nodes("mutation-probe3-results.json")
p4 = nodes("mutation-probe4-results.json")
observed = {k: v for k, v in p2.items() if not k.startswith("m4-")}
observed["m4-smoke-rratio-95"] = p3["m4-smoke-rratio-95"]
observed["m18-provenance-outcomes-drop-tier0"] = p4["m18-provenance-outcomes-drop-tier0"]
assert observed["m0-equivalent-comment"] == [] and p3["m0-equivalent-comment"] == [] and p4["m0-equivalent-comment"] == []
assert all(observed[k] for k in observed if k != "m0-equivalent-comment")
(J / "observed-nodes.json").write_text(json.dumps(observed, indent=1, sort_keys=True) + "\n", encoding="utf-8")
print({k: len(v) for k, v in sorted(observed.items())})
