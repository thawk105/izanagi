def build(expected):
    out = []
    for mid, category, reps in MUTATIONS:
        nodes = [] if expected is None else sorted(expected.get(mid, []))
        out.append({
            "id": mid,
            "category": category,
            "replacements": [{"file": f, "old": o, "new": n} for f, o, n in reps],
            "expected_nodes": nodes,
            "expected_status": "SURVIVED" if not nodes else "KILLED",
            "hang_risk": False,
        })
    return {
        "schema": "izanagi-dev-wave-mutation-spec/v1",
        "estimated_run_seconds": 240,
        "timeout_seconds": 10800,
        "hang_timeout_seconds": 3000,
        "mutations": out,
    }


def main() -> int:
    mode = sys.argv[1] if len(sys.argv) > 1 else "probe"
    if mode == "list":
        print(json.dumps([[m, [[f, o, n] for f, o, n in r]] for m, _, r in MUTATIONS], ensure_ascii=False))
        return 0
    if mode == "probe":
        spec, target = build(None), J / "mutation-spec-probe.json"
    elif mode == "probe-subset":
        ids = set(sys.argv[2:])
        spec = build(None)
        spec["mutations"] = [m for m in spec["mutations"] if m["id"] in ids]
        target = J / "mutation-spec-probe-subset.json"
    else:
        observed = json.loads((J / "observed-nodes.json").read_text(encoding="utf-8"))
        spec, target = build(observed), J / "mutation-spec-final.json"
    text = json.dumps(spec, ensure_ascii=False, indent=1) + "\n"
    target.write_text(text, encoding="utf-8")
    print(target, hashlib.sha256(text.encode("utf-8")).hexdigest(), len(spec["mutations"]))
    return 0


if __name__ == "__main__":
    sys.exit(main())
