#!/usr/bin/env python3
"""[T-2797] (T-2632 の同名 script の写し) insight の逐語の可逆最小正規化 (DW-S07): 行末の space / tab と、ファイル末尾の余分な空行だけを除去する。

usage: normalize_trailing_ws.py <insight-root> <record.md> <relative path>...
各 file について原文 sha256・bytes、正規化後 sha256・bytes、除去位置 (行番号: 除去 bytes の repr) を record.md に書く。
復元は記録どおりの bytes を該当行末へ戻し、末尾空行を足し戻す。可視文字は変えない。
"""
import hashlib
import pathlib
import sys


def normalize(raw: bytes):
    lines = raw.split(b"\n")
    had_final_newline = raw.endswith(b"\n")
    if had_final_newline:
        lines = lines[:-1]
    removed = []
    out = []
    for i, line in enumerate(lines, 1):
        stripped = line.rstrip(b" \t")
        if stripped != line:
            removed.append((i, line[len(stripped):]))
        out.append(stripped)
    eof_blank = 0
    while out and out[-1] == b"":
        out.pop()
        eof_blank += 1
    body = b"\n".join(out) + (b"\n" if out else b"")
    return body, removed, eof_blank, had_final_newline


def main(argv):
    root = pathlib.Path(argv[1])
    record = pathlib.Path(argv[2])
    parts = ["# verbatim・log の可逆最小正規化 (行末空白と末尾空行の除去)\n",
             "`git diff --check` に抵触するため、可視文字を変えずに行末の空白 (space / tab) とファイル末尾の余分な空行だけを除去した。",
             "復元は下表の行番号の末尾に記録どおりの bytes を戻し、末尾空行を記録数だけ足し戻す (元が末尾改行を持つかも記録)。\n"]
    for rel in argv[3:]:
        path = root / rel
        raw = path.read_bytes()
        body, removed, eof_blank, had_nl = normalize(raw)
        path.write_bytes(body)
        parts.append(f"## {rel}\n")
        parts.append(f"- 原文 sha256: `{hashlib.sha256(raw).hexdigest()}`、原文 bytes: {len(raw)}、原文の末尾改行: {'あり' if had_nl else 'なし'}")
        parts.append(f"- 正規化後 sha256: `{hashlib.sha256(body).hexdigest()}`、bytes: {len(body)}、行末空白を除去した行数: {len(removed)}、"
                     f"除去した末尾空行: {eof_blank}")
        if removed:
            parts.append("- 除去位置 (行番号: 除去 bytes の repr):")
            for n, b in removed:
                parts.append(f"  - {n}: `{b!r}`")
        parts.append("")
    record.write_text("\n".join(parts) + "\n", encoding="utf-8")
    print(f"normalized {len(argv) - 3} files")


if __name__ == "__main__":
    main(sys.argv)
